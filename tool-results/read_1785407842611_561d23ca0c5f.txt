   121→  // Centralized build function. Aborts any in-flight build first.
   122→  // Accepts an optional explicit mission to avoid stale-closure bugs (e.g., retry).
   123→  // Uses resultRef instead of result state to avoid being re-created on every build.
   124→  const build = useCallback(async (explicitMission?: string) => {
   125→    const m = (explicitMission ?? mission).trim()
   126→    if (!m) {
   127→      toast.error('Describe what to build first')
   128→      return
   129→    }
   130→
   131→    // If an explicit mission was passed, sync the textarea state
   132→    if (explicitMission && explicitMission !== mission) {
   133→      setMission(explicitMission)
   134→    }
   135→
   136→    // Abort any in-flight build (covers: rebuild, history-click-during-build, reset-during-build)
   137→    abortRef.current?.abort()
   138→    const controller = new AbortController()
   139→    abortRef.current = controller
   140→
   141→    setLoading(true)
   142→    setError(null)
   143→    setFailedMission(null)
   144→    setChatMessages([])
   145→
   146→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   147→    const steps = extractStepsFromMission(m)
   148→    setBuildSteps(steps)
   149→    setPlanSummary(null)
   150→    setLivePreviewHtml(null)
   151→
   152→    // Helper: set error state consistently (replaces 6 repeated blocks)
   153→    const fail = (msg: string) => {
   154→      setError(msg)
   155→      setFailedMission(m)
   156→      if (!resultRef.current) toast.error(msg)
   157→    }
   158→
   159→    try {
   160→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   161→      // This is a separate, fast call (~2-3s). The plan lets us show
   162→      // AUTHENTIC steps to the user before the coder even starts.
   163→      const archRes = await fetch('/api/build/architect', {
   164→        method: 'POST',
   165→        headers: { 'Content-Type': 'application/json' },
   166→        body: JSON.stringify({ mission: m }),
   167→        signal: controller.signal,
   168→      })
   169→
   170→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   171→      const archContentType = archRes.headers.get('content-type') ?? ''
   172→      if (archContentType.includes('application/json')) {
   173→        try { archData = await archRes.json() } catch {}
   174→      }
   175→
   176→      // If architect succeeded, update steps with the REAL plan
   177→      if (archData?.ok && archData.plan) {
   178→        const planSteps = extractStepsFromPlan(archData.plan, m)
   179→        const summary = getPlanSummary(archData.plan)
   180→        setBuildSteps(planSteps)
   181→        setPlanSummary(summary)
   182→        console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   183→      }
   184→      // If architect failed, continue with mission-based steps (already set)
   185→
   186→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   187→      // The route returns Server-Sent Events: progress events while LLM works,
   188→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   189→      const codeRes = await fetch('/api/build/code', {
   190→        method: 'POST',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null }),
   193→        signal: controller.signal,
   194→      })
   195→
   196→      if (!codeRes.ok) {
   197→        // Non-SSE error (400, 429, 413) — parse as JSON
   198→        let errorMsg = `Server error (${codeRes.status})`
   199→        try {
   200→          const errData = await codeRes.json()
   201→          if (errData?.error) errorMsg = errData.error
   202→        } catch {}
   203→        fail(errorMsg)
   204→        return
   205→      }
   206→
   207→      // Read SSE stream
   208→      const reader = codeRes.body?.getReader()
   209→      if (!reader) {
   210→        fail('No response stream')
   211→        return
   212→      }
   213→
   214→      const decoder = new TextDecoder()
   215→      let buffer = ''
   216→      let finalHtml = ''
   217→      let finalTokens = 0
   218→      let finalMs = 0
   219→      let finalQuality = 0
   220→      let finalMetrics = ''
   221→      let streamError: string | null = null
   222→
   223→      while (true) {
   224→        const { done, value } = await reader.read()
   225→        if (done) break
   226→
   227→        buffer += decoder.decode(value, { stream: true })
   228→
   229→        // Parse SSE events (separated by \n\n)
   230→        const events = buffer.split('\n\n')
   231→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   232→
   233→        for (const eventStr of events) {
   234→          const dataLine = eventStr.trim()
   235→          if (!dataLine.startsWith('data: ')) continue
   236→
   237→          try {
   238→            const evt = JSON.parse(dataLine.slice(6))
   239→
   240→            if (evt.type === 'progress') {
   241→              // Update thinking steps with real progress from server
   242→              setBuildSteps(prev => {
   243→                // Replace with server-provided step, keep last as fallback
   244→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   245→                return [prev[0], evt.step, ...prev.slice(2)]
   246→              })
   247→            } else if (evt.type === 'token') {
   248→              // REAL TOKEN STREAMING — accumulate partial HTML for live preview
   249→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   250→              setBuildSteps(prev => {
   251→                const last = prev[prev.length - 1]
   252→                const totalLen = (evt.length ?? 0)
   253→                const newLast = `Generating: ${totalLen} chars...`
   254→                if (last && last.startsWith('Generating:')) {
   255→                  return [...prev.slice(0, -1), newLast]
   256→                }
   257→                return [...prev, newLast]
   258→              })
   259→            } else if (evt.type === 'result') {
   260→              finalHtml = evt.html ?? ''
   261→              finalTokens = evt.tokens ?? 0
   262→              finalMs = evt.ms ?? 0
   263→              finalQuality = evt.quality ?? 0
   264→              finalMetrics = evt.metrics ?? ''
   265→            } else if (evt.type === 'error') {
   266→              streamError = evt.error ?? 'Unknown error'
   267→            }
   268→          } catch {
   269→            // Ignore malformed events
   270→          }
   271→        }
   272→      }
   273→
   274→      if (streamError) {
   275→        fail(streamError)
   276→        return
   277→      }
   278→
   279→      if (!finalHtml) {
   280→        fail('Server returned empty HTML')
   281→        return
   282→      }
   283→
   284→      const buildResult: BuildResult = {
   285→        id: newBuildId(),
   286→        html: finalHtml,
   287→        tokens: finalTokens,
   288→        ms: finalMs,
   289→        mission: m,
   290→      }
   291→
   292→      setResult(buildResult)
   293→
   294→      // Functional setState — avoids stale closure on rapid successive builds
   295→      setHistory(prev => {
   296→        // Dedupe by mission (keep only the latest build per mission)
   297→        const next = [buildResult, ...prev.filter(h => h.mission !== m)].slice(0, 10)
   298→        // Best-effort localStorage; shrink if quota exceeded
   299→        let savedCount = next.length
   300→        try {
   301→          localStorage.setItem('nova_history', JSON.stringify(next))
   302→        } catch (quotaErr) {
   303→          console.error('[NOVA] localStorage quota exceeded:', quotaErr)
   304→          savedCount = 0
   305→          for (let i = next.length - 1; i >= 0; i--) {
   306→            try {
   307→              localStorage.setItem('nova_history', JSON.stringify(next.slice(0, i + 1)))
   308→              savedCount = i + 1
   309→              break
   310→            } catch {
   311→              // keep trying smaller
   312→            }
   313→          }
   314→        }
   315→        // Warn if we couldn't save everything
   316→        if (savedCount < next.length) {
   317→          toast.error(`localStorage full — only ${savedCount} of ${next.length} builds saved to history`)
   318→        }
   319→        return next
   320→      })
   321→
   322→      toast.success(`Built in ${(finalMs / 1000).toFixed(1)}s · ${finalTokens} tokens · quality: ${finalQuality}`)
   323→      setQualityScore(finalQuality)
   324→      setQualityMetrics(finalMetrics)
   325→    } catch (err: unknown) {
   326→      // AbortError = user started a new build, loaded history, or navigated away; silently ignore
   327→      if (err instanceof DOMException && err.name === 'AbortError') return
   328→      const msg = err instanceof Error ? err.message : 'Network error'
   329→      fail(msg)
   330→    } finally {
   331→      // Only clear loading if this controller is still the active one
   332→      if (abortRef.current === controller) {
   333→        abortRef.current = null
   334→        setLoading(false)
   335→      }
   336→    }
   337→  }, [mission])
   338→
   339→  const loadFromHistory = useCallback((h: BuildResult) => {
   340→    // Abort any in-flight build or refine
   341→    abortRef.current?.abort()
   342→    abortRef.current = null
   343→    refineAbortRef.current?.abort()
   344→    refineAbortRef.current = null
   345→    setLoading(false)
   346→    setRefining(false)
   347→    setResult(h)
   348→    setMission(h.mission)
   349→    setError(null)
   350→    setFailedMission(null)
   351→    setChatMessages([])
   352→  }, [])
   353→
   354→  const cancelBuild = useCallback(() => {
   355→    // Cancel only aborts the in-flight build — does NOT clear mission or result.
   356→    // The user keeps their mission text and the old preview (if any).
   357→    abortRef.current?.abort()
   358→    abortRef.current = null
   359→    setLoading(false)
   360→    setError(null)
   361→    setFailedMission(null)
   362→  }, [])
   363→
   364→  const reset = useCallback(() => {
   365→    // Reset clears everything — used by the "New" button.
   366→    abortRef.current?.abort()
   367→    abortRef.current = null
   368→    setLoading(false)
   369→    setRefining(false)
   370→    setResult(null)
   371→    setError(null)
   372→    setFailedMission(null)
   373→    setMission('')
   374→    setChatMessages([])
   375→    setQualityScore(0)
   376→    setQualityMetrics('')
   377→    setLivePreviewHtml(null)
   378→  }, [])
   379→
   380→  const retryFailed = useCallback(() => {
   381→    // If the user edited the textarea after the failure, use the edited mission.
   382→    // Otherwise, use the original failed mission.
   383→    const currentMission = mission.trim()

   385→      build(currentMission)
   386→    } else if (failedMission) {
   387→      build(failedMission)
   388→    }
   389→  }, [failedMission, build, mission])
   390→
   391→  const download = useCallback(() => {
   392→    if (!result?.html) return
   393→    const blob = new Blob([result.html], { type: 'text/html' })
   394→    const url = URL.createObjectURL(blob)
   395→    const a = document.createElement('a')
   396→    a.href = url
   397→    const filename = sanitizeFilename(result.mission)
   398→    a.download = filename
   399→    document.body.appendChild(a)
   400→    a.click()
   401→    document.body.removeChild(a)
   402→    URL.revokeObjectURL(url)
   403→    toast.success(`Downloaded ${filename}`)
   404→  }, [result])
   405→
   406→  // Copy HTML to clipboard
   407→  const copyHtml = useCallback(async () => {
   408→    if (!result?.html) return
   409→    try {
   410→      await navigator.clipboard.writeText(result.html)
   411→      toast.success('HTML copied to clipboard')
   412→    } catch {
   413→      toast.error('Failed to copy — try downloading instead')
   414→    }
   415→  }, [result])
   416→
   417→  // Open the HTML in a new browser tab (full-screen preview)
   418→  const openInNewTab = useCallback(() => {
   419→    if (!result?.html) return
   420→    const blob = new Blob([result.html], { type: 'text/html' })
   421→    const url = URL.createObjectURL(blob)
   422→    window.open(url, '_blank', 'noopener,noreferrer')
   423→    // Revoke after 30s — enough time for the tab to load
   424→    setTimeout(() => URL.revokeObjectURL(url), 30_000)
   425→    toast.info('Opened in new tab')
   426→  }, [result])
   427→
   428→  // Chat refine: SSE streaming — same pattern as build/code
   429→  const sendChat = useCallback(async () => {
   430→    const msg = chatInput.trim()
   431→    const currentResult = resultRef.current
   432→    if (!msg || refining || !currentResult) return
   433→
   434→    const userMsg: ChatMessage = { role: 'user', content: msg, ts: Date.now() }
   435→    setChatMessages(prev => [...prev, userMsg])
   436→    setChatInput('')
   437→    setRefining(true)
   438→
   439→    refineAbortRef.current?.abort()
   440→    const controller = new AbortController()
   441→    refineAbortRef.current = controller
   442→
   443→    // Clear live preview so refine starts fresh
   444→    setLivePreviewHtml(null)
   445→
   446→    try {
   447→      const res = await fetch('/api/refine', {
   448→        method: 'POST',
   449→        headers: { 'Content-Type': 'application/json' },
   450→        body: JSON.stringify({ mission: currentResult.mission, html: currentResult.html, message: msg }),
   451→        signal: controller.signal,
   452→      })
   453→
   454→      if (!res.ok) {
   455→        let errorMsg = `Server error (${res.status})`
   456→        try { const e = await res.json(); if (e?.error) errorMsg = e.error } catch {}
   457→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${errorMsg}`, ts: Date.now() }])
   458→        toast.error(errorMsg)
   459→        return
   460→      }
   461→
   462→      // Read SSE stream
   463→      const reader = res.body?.getReader()
   464→      if (!reader) {
   465→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: No stream', ts: Date.now() }])
   466→        return
   467→      }
   468→
   469→      const decoder = new TextDecoder()
   470→      let buffer = ''
   471→      let finalHtml = ''
   472→      let finalTokens = 0
   473→      let finalMs = 0
   474→      let finalQuality = 0
   475→      let finalMetrics = ''
   476→      let streamError: string | null = null
   477→
   478→      while (true) {
   479→        const { done, value } = await reader.read()
   480→        if (done) break
   481→
   482→        buffer += decoder.decode(value, { stream: true })
   483→        const events = buffer.split('\n\n')
   484→        buffer = events.pop() ?? ''
   485→
   486→        for (const eventStr of events) {
   487→          const dataLine = eventStr.trim()
   488→          if (!dataLine.startsWith('data: ')) continue
   489→          try {
   490→            const evt = JSON.parse(dataLine.slice(6))
   491→            if (evt.type === 'token') {
   492→              // Live token streaming — show refined HTML appearing in real-time
   493→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   494→            } else if (evt.type === 'result') {
   495→              finalHtml = evt.html ?? ''
   496→              finalTokens = evt.tokens ?? 0
   497→              finalMs = evt.ms ?? 0
   498→              finalQuality = evt.quality ?? 0
   499→              finalMetrics = evt.metrics ?? ''
   500→            } else if (evt.type === 'error') {
   501→              streamError = evt.error ?? 'Unknown error'
   502→            }
   503→          } catch {}
   504→        }
   505→      }
   506→
   507→      if (streamError) {
   508→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${streamError}`, ts: Date.now() }])
   509→        toast.error(streamError)
   510→        return
   511→      }
   512→
   513→      if (!finalHtml) {
   514→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: Empty response', ts: Date.now() }])
   515→        return
   516→      }
   517→
   518→      const refinedResult: BuildResult = {
   519→        ...currentResult,
   520→        id: newBuildId(),
   521→        html: finalHtml,
   522→      }
   523→      setResult(refinedResult)
   524→
   525→      setHistory(prev => {
   526→        const next = [refinedResult, ...prev.filter(h => h.mission !== currentResult.mission)].slice(0, 10)
   527→        try { localStorage.setItem('nova_history', JSON.stringify(next)) } catch {}
   528→        return next
   529→      })
   530→
   531→      setChatMessages(prev => [...prev, {
   532→        role: 'assistant',
   533→        content: `Updated! ${finalMs ? `${(finalMs / 1000).toFixed(1)}s` : ''} · ${finalTokens} tokens`,
   534→        ts: Date.now(),
   535→      }])
   536→
   537→      toast.success(`Refined! ${(finalMs / 1000).toFixed(1)}s · quality: ${finalQuality}`)
   538→      setQualityScore(finalQuality)
   539→      setQualityMetrics(finalMetrics)
   540→    } catch (err: unknown) {
   541→      if (err instanceof DOMException && err.name === 'AbortError') return
   542→      const failMsg = err instanceof Error ? err.message : 'Network error'
   543→      setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${failMsg}`, ts: Date.now() }])
   544→      toast.error(failMsg)
   545→    } finally {
   546→      if (refineAbortRef.current === controller) {
   547→        refineAbortRef.current = null
   548→        setRefining(false)
   549→        setLivePreviewHtml(null) // clear live preview so final result shows
   550→      }
   551→    }
   552→  }, [chatInput, refining])
   553→
   554→  // Auto-scroll chat to bottom on new messages
   555→  useEffect(() => {
   556→    if (chatScrollRef.current) {
   557→      chatScrollRef.current.scrollTop = chatScrollRef.current.scrollHeight
   558→    }
   559→  }, [chatMessages, refining])
   560→
   561→  // Keyboard shortcuts: Esc=cancel build/refine, ⌘S/Ctrl+S=download, ⌘N/Ctrl+N=new
   562→  useEffect(() => {
   563→    const onKey = (e: KeyboardEvent) => {
   564→      // Esc cancels a build or refine
