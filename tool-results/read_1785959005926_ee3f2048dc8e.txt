     1→// Golden Templates — pre-built HTML app templates that NOVA uses as seeds.
     2→//
     3→// When a user's mission closely matches a known template (e.g. "build a snake game"),
     4→// NOVA can include the template HTML as a starting point in the LLM prompt. This:
     5→// - Reduces generation time (LLM has a working baseline to refine)
     6→// - Improves output quality (templates are hand-crafted, working apps)
     7→// - Lowers token usage (LLM doesn't start from scratch)
     8→//
     9→// Templates are COMPLETE, WORKING HTML files with NOVA's dark theme. Each has a
    10→// list of keywords used to match against user missions.
    11→//
    12→// Pure module — no I/O, no LLM calls. The `html` strings are static.
    13→
    14→// ── Types ──
    15→
    16→/** A pre-built HTML app template. */
    17→export interface GoldenTemplate {
    18→  /** Unique identifier, e.g. "snake-game". */
    19→  id: string
    20→  /** Human-readable name, e.g. "Snake Game". */
    21→  name: string
    22→  /** Keywords that, if present in the mission, suggest this template. Lowercase. */
    23→  keywords: string[]
    24→  /** Short description of what the template provides. */
    25→  description: string
    26→  /** Complete working HTML document (dark theme, single-file). */
    27→  html: string
    28→}

   186→    snake.forEach(function (s, i) {
   187→      ctx.fillStyle = i === 0 ? '#22d3ee' : '#3b82f6';

   193→    if (!running || paused) return;

   196→
   197→    // Wall collision
   198→    if (head.x < 0 || head.x >= CELLS || head.y < 0 || head.y >= CELLS) {
   199→      return gameOver();
   200→    }
   201→    // Self collision
   202→    if (snake.some(function (s) { return s.x === head.x && s.y === head.y; })) {

   834→// ── Template matching ──
   835→
   836→/**
   837→ * Find the best-matching golden template for a user mission.
   838→ * Scores each template by counting keyword matches (word-boundary aware)
   839→ * against the lowercased mission. Returns the highest-scoring template,
   840→ * or null if no template has at least one keyword match.
   841→ *
   842→ * Word-boundary matching prevents false positives (e.g. "css" matching "basilisk"
   843→ * or "todo" matching "autodo"). Multi-word keywords like "snake game" match if
   844→ * the literal phrase appears in the mission.
   845→ */
   846→export function findTemplate(mission: string): GoldenTemplate | null {
   847→  if (!mission || !mission.trim()) return null
   848→  const lower = ' ' + mission.toLowerCase() + ' '
   849→
   850→  let best: GoldenTemplate | null = null
   851→  let bestScore = 0
   852→
   853→  for (const tpl of GOLDEN_TEMPLATES) {
   854→    let score = 0
   855→    for (const kw of tpl.keywords) {
   856→      const k = kw.toLowerCase()
   857→      if (k.includes(' ')) {
   858→        // Multi-word keyword: literal substring match
   859→        if (lower.includes(k)) score += 3
   860→      } else {
   861→        // Single-word keyword: match on word boundaries
   862→        // \b in JS regex works with ASCII word chars; keywords here are all ASCII.
   863→        const re = new RegExp('\\b' + escapeRegex(k) + '\\b')
   864→        if (re.test(lower)) score += 2
   865→      }
   866→    }
   867→    if (score > bestScore) {
   868→      bestScore = score
   869→      best = tpl
   870→    }
   871→  }
   872→
   873→  // Require at least one match (score >= 2 for single-word, >= 3 for multi-word)
   874→  return bestScore >= 2 ? best : null
   875→}
   876→
   877→function escapeRegex(s: string): string {
   878→  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   879→}
