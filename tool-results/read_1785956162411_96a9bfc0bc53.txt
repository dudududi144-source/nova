    57→    57→function makeResponse(opts: MockResponseInit): Response {
    58→    58→  const status = opts.status ?? 200
    59→    59→  // Explicit null body — Response(null) creates a Response with body=null.
    60→    60→  if (opts.body === null) {
    61→    61→    return new Response(null, { status })
    62→    62→  }
    63→    63→  if (opts.body) {
    64→    64→    return new Response(opts.body, { status })
    65→    65→  }
    66→    66→  if (opts.json) {
    67→    67→    const body = JSON.stringify(opts.json())
    68→    68→    return new Response(body, { status, headers: { 'Content-Type': 'application/json' } })
    69→    69→  }
    70→    70→  if (opts.text !== undefined) {
    71→    71→    return new Response(opts.text, { status })
    72→    72→  }
    73→    73→  return new Response('', { status })
    74→    74→}

    76→    76→function makeSseBody(events: Array<string>): ReadableStream<Uint8Array> {
    77→    77→  const encoder = new TextEncoder()
    78→    78→  return new ReadableStream<Uint8Array>({
    79→    79→    start(controller) {
    80→    80→      for (const e of events) controller.enqueue(encoder.encode(e))
    81→    81→      controller.close()
    82→    82→    },
    83→    83→  })
    84→    84→}
