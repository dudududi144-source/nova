    88→  // Save history to localStorage — pure side effect, called OUTSIDE of setState updaters
    89→  // to avoid double-firing in React StrictMode.
    90→  const saveHistoryToStorage = useCallback((items: BuildResult[]) => {
    91→    let savedCount = items.length
    92→    try {
    93→      localStorage.setItem('nova_history', JSON.stringify(items))
    94→    } catch (quotaErr) {
    95→      console.error('[NOVA] localStorage quota exceeded:', quotaErr)
    96→      savedCount = 0
    97→      for (let i = items.length - 1; i >= 0; i--) {
    98→        try {
    99→          localStorage.setItem('nova_history', JSON.stringify(items.slice(0, i + 1)))
   100→          savedCount = i + 1
   101→          break
   102→        } catch {
   103→          // keep trying smaller
   104→        }
   105→      }
   106→    }
   107→    if (savedCount < items.length) {

   114→  const addBuildToHistory = useCallback((buildResult: BuildResult) => {

   124→  // Abort any in-flight build AND refine on unmount

   131→
   132→  // Ref mirror of buildSteps so the elapsed-time effect doesn't need buildSteps in its deps.
   133→  // If buildSteps is in the deps, the effect tears down and re-runs on every token event
   134→  // (because setBuildSteps is called on every token), resetting startTime and thinkingStep.
   135→  const buildStepsRef = useRef<string[]>(['Building...'])
   136→  useEffect(() => {
   137→    buildStepsRef.current = buildSteps
   138→  }, [buildSteps])
   139→
   140→  // Elapsed time counter + dynamic thinking step rotation.
   141→  // CRITICAL: deps are [loading, refining] ONLY — not buildSteps.
   142→  // If buildSteps were in deps, every token event would reset startTime to Date.now(),
   143→  // the interval would be cleared before it ever fires, and elapsed would stay at 0.
   144→  useEffect(() => {
   145→    if (!loading && !refining) {
   146→      setElapsed(0)
   147→      setThinkingStep(0)
   148→      return
   149→    }
   150→    const startTime = Date.now()
   151→    let step = 0
   152→    setThinkingStep(0)
   153→
   154→    const timer = setInterval(() => {
   155→      const sec = Math.floor((Date.now() - startTime) / 1000)
   156→      setElapsed(sec)
   157→
   158→      // Read steps from ref (not from closure — closure would be stale)
   159→      const steps = loading ? buildStepsRef.current : REFINE_THINKING_STEPS
   160→      if (steps.length === 0) return
   161→
   162→      // Rotate through steps — ~4s per step for build, 5s for refine
   163→      const stepDuration = loading ? 4 : 5
   164→      const nextStep = Math.min(steps.length - 1, Math.floor(sec / stepDuration))
   165→      if (nextStep !== step) {
   166→        step = nextStep
   167→        setThinkingStep(step)
   168→      }
   169→    }, 1000)
   170→    return () => clearInterval(timer)
   171→  }, [loading, refining])
   172→
   173→  // Centralized build function. Aborts any in-flight build first.
   174→  // Accepts an optional explicit mission to avoid stale-closure bugs (e.g., retry).
   175→  // Uses resultRef instead of result state to avoid being re-created on every build.
   176→  const build = useCallback(async (explicitMission?: string) => {
   177→    const m = (explicitMission ?? mission).trim()
   178→    if (!m) {
   179→      toast.error('Describe what to build first')
   180→      return
   181→    }
   182→
   183→    // If an explicit mission was passed, sync the textarea state
   184→    if (explicitMission && explicitMission !== mission) {
   185→      setMission(explicitMission)
   186→    }
   187→
   188→    // Abort any in-flight build AND refine (covers: rebuild during refine, history-click-during-build, reset-during-build)
   189→    // This prevents the race condition where build and refine run simultaneously and
   190→    // both call setResult/setHistory, corrupting state.
   191→    abortRef.current?.abort()
   192→    refineAbortRef.current?.abort()
   193→    refineAbortRef.current = null
   194→    setRefining(false)
   195→    const controller = new AbortController()
   196→    abortRef.current = controller
   197→
   198→    setLoading(true)
   199→    setError(null)
   200→    setFailedMission(null)
   201→    setChatMessages([])
   202→
   203→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   204→    const steps = extractStepsFromMission(m)
   205→    setBuildSteps(steps)
   206→    setPlanSummary(null)
   207→    setLivePreviewHtml(null)
   208→
   209→    // Helper: set error state consistently (replaces 6 repeated blocks)
   210→    const fail = (msg: string) => {
   211→      setError(msg)
   212→      setFailedMission(m)
   213→      if (!resultRef.current) toast.error(msg)
   214→    }
   215→
   216→    try {
   217→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   218→      // This is a separate, fast call (~2-3s). The plan lets us show
   219→      // AUTHENTIC steps to the user before the coder even starts.
   220→      const archRes = await fetch('/api/build/architect', {
   221→        method: 'POST',
   222→        headers: { 'Content-Type': 'application/json' },
   223→        body: JSON.stringify({ mission: m }),
   224→        signal: controller.signal,
   225→      })
   226→
   227→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   228→      const archContentType = archRes.headers.get('content-type') ?? ''
   229→      if (archContentType.includes('application/json')) {
   230→        try { archData = await archRes.json() } catch {}
   231→      }
   232→
   233→      // If architect succeeded, update steps with the REAL plan
   234→      if (archData?.ok && archData.plan) {
   235→        const planSteps = extractStepsFromPlan(archData.plan, m)
   236→        const summary = getPlanSummary(archData.plan)
   237→        setBuildSteps(planSteps)
   238→        setPlanSummary(summary)
   239→        console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   240→      }
   241→      // If architect failed, continue with mission-based steps (already set)
   242→
   243→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   244→      // The route returns Server-Sent Events: progress events while LLM works,
   245→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   246→      const codeRes = await fetch('/api/build/code', {
   247→        method: 'POST',
   248→        headers: { 'Content-Type': 'application/json' },
   249→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null }),
   250→        signal: controller.signal,
   251→      })
   252→
   253→      if (!codeRes.ok) {
   254→        // Non-SSE error (400, 429, 413) — parse as JSON
   255→        let errorMsg = `Server error (${codeRes.status})`
   256→        try {
   257→          const errData = await codeRes.json()
   258→          if (errData?.error) errorMsg = errData.error
   259→        } catch {}
   260→        fail(errorMsg)
   261→        return
   262→      }
   263→
   264→      // Read SSE stream
   265→      const reader = codeRes.body?.getReader()
   266→      if (!reader) {
   267→        fail('No response stream')
   268→        return
   269→      }
   270→
   271→      const decoder = new TextDecoder()
   272→      let buffer = ''
   273→      let finalHtml = ''
   274→      let finalTokens = 0
   275→      let finalMs = 0
   276→      let finalQuality = 0
   277→      let finalMetrics = ''
   278→      let streamError: string | null = null
   279→
   280→      while (true) {
   281→        const { done, value } = await reader.read()
   282→        if (done) break
   283→
   284→        buffer += decoder.decode(value, { stream: true })
   285→
   286→        // Parse SSE events (separated by \n\n or \r\n\r\n — some proxies normalize to \r\n)
   287→        // Normalize \r\n to \n first so the split works regardless of proxy behavior.
   288→        const normalized = buffer.replace(/\r\n/g, '\n')
   289→        const events = normalized.split('\n\n')
   290→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   291→
   292→        for (const eventStr of events) {
   293→          const dataLine = eventStr.trim()
   294→          if (!dataLine.startsWith('data: ')) continue
   295→
   296→          try {
   297→            const evt = JSON.parse(dataLine.slice(6))
   298→
   299→            if (evt.type === 'progress') {
   300→              // Update thinking steps with real progress from server
   301→              setBuildSteps(prev => {
   302→                // Replace with server-provided step, keep last as fallback
   303→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   304→                return [prev[0], evt.step, ...prev.slice(2)]
   305→              })
   306→            } else if (evt.type === 'token') {
   307→              // REAL TOKEN STREAMING — accumulate partial HTML for live preview
   308→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   309→              setBuildSteps(prev => {
   310→                const last = prev[prev.length - 1]
   311→                const totalLen = (evt.length ?? 0)
   312→                const newLast = `Generating: ${totalLen} chars...`
   313→                if (last && last.startsWith('Generating:')) {
   314→                  return [...prev.slice(0, -1), newLast]
   315→                }
   316→                return [...prev, newLast]
   317→              })
   318→            } else if (evt.type === 'result') {
   319→              finalHtml = evt.html ?? ''
   320→              finalTokens = evt.tokens ?? 0
   321→              finalMs = evt.ms ?? 0
   322→              finalQuality = evt.quality ?? 0
   323→              finalMetrics = evt.metrics ?? ''
   324→            } else if (evt.type === 'error') {
   325→              streamError = evt.error ?? 'Unknown error'
   326→            }
   327→          } catch {
   328→            // Ignore malformed events
   329→          }
   330→        }
   331→      }
   332→
   333→      if (streamError) {
   334→        fail(streamError)
   335→        return
   336→      }
   337→
   338→      if (!finalHtml) {
   339→        fail('Server returned empty HTML')
   340→        return
   341→      }
   342→
   343→      const buildResult: BuildResult = {
   344→        id: newBuildId(),
   345→        html: finalHtml,
   346→        tokens: finalTokens,
   347→        ms: finalMs,
   348→        mission: m,
   349→      }
   350→
   351→      setResult(buildResult)
   352→      resultRef.current = buildResult // Update ref synchronously
   353→      addBuildToHistory(buildResult)
   354→
   355→      toast.success(`Built in ${(finalMs / 1000).toFixed(1)}s · ${finalTokens} tokens · quality: ${finalQuality}`)
   356→      setQualityScore(finalQuality)
   357→      setQualityMetrics(finalMetrics)
   358→    } catch (err: unknown) {
   359→      // AbortError = user started a new build, loaded history, or navigated away; silently ignore
   360→      if (err instanceof DOMException && err.name === 'AbortError') return
   361→      const msg = err instanceof Error ? err.message : 'Network error'
   362→      fail(msg)
   363→    } finally {
   364→      // Only clear loading if this controller is still the active one
   365→      if (abortRef.current === controller) {
   366→        abortRef.current = null
   367→        setLoading(false)
   368→      }
   369→    }
   370→  }, [mission])
   371→
   372→  const loadFromHistory = useCallback((h: BuildResult) => {
   373→    // Abort any in-flight build or refine
   374→    abortRef.current?.abort()
   375→    abortRef.current = null
   376→    refineAbortRef.current?.abort()
   377→    refineAbortRef.current = null
   378→    setLoading(false)
   379→    setRefining(false)
   380→    setResult(h)
   381→    resultRef.current = h // Update ref synchronously so sendChat sees the correct result immediately
   382→    setMission(h.mission)
   383→    setError(null)
   384→    setFailedMission(null)
   385→    setChatMessages([])
   386→    // Reset all derived state so we don't show the previous build's badges/plan
   387→    setQualityScore(0)
   388→    setQualityMetrics('')
   389→    setPlanSummary(null)
   390→    setLivePreviewHtml(null)
   391→    setConfirmClear(false)
   392→  }, [])
   393→
   394→  const cancelBuild = useCallback(() => {
   395→    // Cancel only aborts the in-flight build — does NOT clear mission or result.
   396→    // The user keeps their mission text and the old preview (if any).
   397→    abortRef.current?.abort()
   398→    abortRef.current = null
   399→    setLoading(false)
   400→    setError(null)
   401→    setFailedMission(null)
   402→    setLivePreviewHtml(null) // Clear stale partial HTML so it doesn't flash back later

   404→
   405→  // Cancel an in-flight refine — accessible via the toolbar Cancel button during refining.

   408→  const cancelRefine = useCallback(() => {
   409→    refineAbortRef.current?.abort()
   410→    refineAbortRef.current = null
   411→    setRefining(false)
   412→    setLivePreviewHtml(null)
   413→  }, [])
   414→
   415→  const reset = useCallback(() => {
   416→    // Reset clears everything — used by the "New" button.
   417→    // Abort BOTH in-flight build and refine to prevent phantom state mutations.
   418→    abortRef.current?.abort()
   419→    abortRef.current = null
   420→    refineAbortRef.current?.abort()
   421→    refineAbortRef.current = null
   422→    setLoading(false)
   423→    setRefining(false)
   424→    setResult(null)
   425→    setError(null)
   426→    setFailedMission(null)
   427→    setMission('')
   428→    setChatMessages([])
   429→    setQualityScore(0)
   430→    setQualityMetrics('')
   431→    setLivePreviewHtml(null)
   432→    setPlanSummary(null)
   433→    setConfirmClear(false)
   434→  }, [])
   435→
   436→  const retryFailed = useCallback(() => {
   437→    // If the user edited the textarea after the failure, use the edited mission.
   438→    // Otherwise, use the original failed mission.
   439→    const currentMission = mission.trim()
   440→    if (currentMission && currentMission !== failedMission) {
   441→      build(currentMission)
   442→    } else if (failedMission) {
   443→      build(failedMission)
   444→    }
   445→  }, [failedMission, build, mission])
   446→
   447→  const download = useCallback(() => {
   448→    if (!result?.html) return
   449→    const blob = new Blob([result.html], { type: 'text/html' })
   450→    const url = URL.createObjectURL(blob)
   451→    const a = document.createElement('a')
   452→    a.href = url
   453→    const filename = sanitizeFilename(result.mission)
   454→    a.download = filename
   455→    document.body.appendChild(a)
   456→    a.click()
   457→    document.body.removeChild(a)
   458→    // Delay revocation — Safari <16 and Firefox with large blobs read asynchronously.
   459→    // Revoking immediately can produce 0-byte files.
   460→    setTimeout(() => URL.revokeObjectURL(url), 1000)
   461→    toast.success(`Downloaded ${filename}`)
   462→  }, [result])
   463→
   464→  // Copy HTML to clipboard
   465→  const copyHtml = useCallback(async () => {
   466→    if (!result?.html) return
   467→    try {
   468→      await navigator.clipboard.writeText(result.html)
   469→      toast.success('HTML copied to clipboard')
   470→    } catch {
   471→      toast.error('Failed to copy — try downloading instead')
   472→    }
   473→  }, [result])
   474→
   475→  // Open the HTML in a new browser tab (full-screen preview)
   476→  const openInNewTab = useCallback(() => {
   477→    if (!result?.html) return
   478→    const blob = new Blob([result.html], { type: 'text/html' })
   479→    const url = URL.createObjectURL(blob)
   480→    window.open(url, '_blank', 'noopener,noreferrer')
   481→    // Revoke after 30s — enough time for the tab to load
   482→    setTimeout(() => URL.revokeObjectURL(url), 30_000)
   483→    toast.info('Opened in new tab')
   484→  }, [result])
   485→
   486→  // Chat refine: SSE streaming — same pattern as build/code
   487→  const sendChat = useCallback(async () => {
   488→    const msg = chatInput.trim()
   489→    const currentResult = resultRef.current
   490→    if (!msg || refining) return
   491→    if (!currentResult) {
   492→      toast.info('Build something first, then you can refine it')
   493→      return
   494→    }
   495→
   496→    const userMsg: ChatMessage = { role: 'user', content: msg, ts: Date.now() }
   497→    setChatMessages(prev => [...prev, userMsg])
   498→    // Don't clear chatInput yet — clear it only after the refine succeeds.
   499→    // If the refine fails, we restore the input so the user doesn't lose their message.
   500→    setRefining(true)
   501→
   502→    refineAbortRef.current?.abort()
   503→    const controller = new AbortController()
   504→    refineAbortRef.current = controller
   505→
   506→    // Clear live preview so refine starts fresh
   507→    setLivePreviewHtml(null)
   508→
   509→    try {
   510→      const res = await fetch('/api/refine', {
   511→        method: 'POST',
   512→        headers: { 'Content-Type': 'application/json' },
   513→        body: JSON.stringify({ mission: currentResult.mission, html: currentResult.html, message: msg }),
   514→        signal: controller.signal,
   515→      })
   516→
   517→      if (!res.ok) {
   518→        let errorMsg = `Server error (${res.status})`
   519→        try { const e = await res.json(); if (e?.error) errorMsg = e.error } catch {}
   520→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${errorMsg}`, ts: Date.now() }])
   521→        toast.error(errorMsg)
   522→        return
   523→      }
   524→
   525→      // Read SSE stream
   526→      const reader = res.body?.getReader()
   527→      if (!reader) {
   528→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: No stream', ts: Date.now() }])
   529→        return
   530→      }
   531→
   532→      const decoder = new TextDecoder()
   533→      let buffer = ''
   534→      let finalHtml = ''
   535→      let finalTokens = 0
   536→      let finalMs = 0
   537→      let finalQuality = 0
   538→      let finalMetrics = ''
   539→      let streamError: string | null = null
   540→
   541→      while (true) {
   542→        const { done, value } = await reader.read()
   543→        if (done) break
   544→
   545→        buffer += decoder.decode(value, { stream: true })
   546→        // Normalize \r\n to \n for proxy compatibility, then split on \n\n
   547→        const normalized = buffer.replace(/\r\n/g, '\n')
   548→        const events = normalized.split('\n\n')
   549→        buffer = events.pop() ?? ''
   550→
   551→        for (const eventStr of events) {
   552→          const dataLine = eventStr.trim()
   553→          if (!dataLine.startsWith('data: ')) continue
   554→          try {
