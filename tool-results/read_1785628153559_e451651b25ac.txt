     3→// FileViewer — multi-file code viewer for non-HTML LLM output.
     4→//
     5→// NOVA's primary mode is single-file HTML (previewed in a sandboxed iframe).
     6→// But the LLM sometimes produces multi-file output: a React project (separate
     7→// component files), a Python script with requirements.txt, or a Node.js app
     8→// with package.json. These can't be previewed in an iframe, so we show them
     9→// here with:
    10→//
    11→// - A recursive file tree (collapsible folders) on the left
    12→// - The selected file's content on the right with syntax highlighting
    13→// - Line numbers, char count, and a language badge
    14→// - Per-file actions: copy, download single file
    15→// - Project-level action: download all as ZIP (uses our dependency-free zip.ts)
    16→//
    17→// The syntax highlighter is a custom tokenizer (no external deps — avoids
    18→// pulling in 200KB+ for prismjs/highlight.js). It supports 9 languages:
    19→// HTML, CSS, JavaScript, TypeScript, JSX, Python, JSON, Markdown, Bash.
    20→//
    21→// All UI strings are user-visible (toasts, button labels). The component is
    22→// client-only because it uses browser APIs (clipboard, Blob, URL.createObjectURL)
    23→// and React state.

    61→// ── Syntax highlighting ──
    62→//
    63→// Tokenizer-based syntax highlighter. Returns an array of { type, value }
    64→// tokens which the renderer maps to colored spans.
    65→//
    66→// Token types:
    67→//   'keyword'    — language keywords (if, for, function, etc.)
    68→//   'string'     — string literals (single/double/backtick)
    69→//   'comment'    — comments (// and /* */ and #)
    70→//   'number'     — numeric literals
    71→//   'tag'        — HTML/XML tag names
    72→//   'attr'       — HTML/XML attribute names
    73→//   'punctuation'— brackets, parens, operators
    74→//   'function'   — function call names (heuristic: identifier followed by `(`)
    75→//   'plain'      — default text
    76→//
    77→// The tokenizer is intentionally simple — it doesn't do full AST parsing.
    78→// False positives are fine (better to over-highlight than miss things).
    79→
    80→export type TokenType =
    81→  | 'keyword'
    82→  | 'string'
    83→  | 'comment'
    84→  | 'number'
    85→  | 'tag'
    86→  | 'attr'
    87→  | 'punctuation'
    88→  | 'function'
    89→  | 'plain'
    90→
    91→export interface Token {
    92→  type: TokenType
    93→  value: string
    94→}

   155→/**
   156→ * Tokenize a line of source code for syntax highlighting.
   157→ *
   158→ * The tokenizer processes one line at a time (no multi-line state except for
   159→ * block comments, which are tracked via a flag passed by the caller).
   160→ *
   161→ * @param line The line of code to tokenize.
   162→ * @param language The language identifier (lowercase).
   163→ * @param inBlockComment Whether we're inside a slash-star-star-slash block
   164→ *   comment that started on a previous line. Updated and returned for the
   165→ *   next call.
   166→ */
   167→export function tokenizeLine(
   168→  line: string,
   169→  language: string,
   170→  inBlockComment: boolean = false,
   171→): { tokens: Token[]; inBlockComment: boolean } {
   172→  const lang = (language || 'text').toLowerCase()
   173→  const keywords = KEYWORDS[lang]
   174→  const tokens: Token[] = []
   175→  let i = 0
   176→  let blockComment = inBlockComment
   177→
   178→  // HTML / XML: handle tags, attributes, and embedded content separately.
   179→  if (lang === 'html' || lang === 'xml') {
   180→    return tokenizeHtml(line, blockComment)
   181→  }
   182→
   183→  // CSS: handle selectors, properties, values, and { } ; : ,
   184→  if (lang === 'css') {
   185→    return tokenizeCss(line, blockComment)
   186→  }
   187→
   188→  // JSON: handle keys, strings, numbers, and structural punctuation.
   189→  if (lang === 'json') {
   190→    return tokenizeJson(line)
   191→  }
   192→
   193→  // Markdown: headings, bold, italic, code, links, lists.
   194→  if (lang === 'markdown') {
   195→    return { tokens: tokenizeMarkdown(line), inBlockComment: false }
   196→  }
   197→
   198→  // C-like languages: JavaScript, TypeScript, JSX, TSX, Python, Bash.
   199→  while (i < line.length) {
   200→    // Continue block comment from previous line
   201→    if (blockComment) {
   202→      const endIdx = line.indexOf('*/', i)
   203→      if (endIdx < 0) {
   204→        tokens.push({ type: 'comment', value: line.slice(i) })
   205→        return { tokens, inBlockComment: true }
   206→      }
   207→      tokens.push({ type: 'comment', value: line.slice(i, endIdx + 2) })
   208→      i = endIdx + 2
   209→      blockComment = false
   210→      continue
   211→    }
   212→
   213→    const ch = line[i]
   214→
   215→    // Whitespace — push as plain (preserves indentation)
   216→    if (ch === ' ' || ch === '\t') {
   217→      let j = i + 1
   218→      while (j < line.length && (line[j] === ' ' || line[j] === '\t')) j++
   219→      tokens.push({ type: 'plain', value: line.slice(i, j) })
   220→      i = j
   221→      continue
   222→    }
   223→
   224→    // Line comments
   225→    if (C_LIKE_COMMENTS.has(lang) && ch === '/' && line[i + 1] === '/') {
   226→      tokens.push({ type: 'comment', value: line.slice(i) })
   227→      i = line.length
   228→      continue
   229→    }
   230→    if (lang === 'python' && ch === '#') {
   231→      tokens.push({ type: 'comment', value: line.slice(i) })
   232→      i = line.length
   233→      continue
   234→    }
   235→    if (lang === 'bash' && ch === '#') {
   236→      tokens.push({ type: 'comment', value: line.slice(i) })
   237→      i = line.length
   238→      continue
   239→    }
   240→
   241→    // Block comment start (C-like)
   242→    if (C_LIKE_COMMENTS.has(lang) && ch === '/' && line[i + 1] === '*') {
   243→      const endIdx = line.indexOf('*/', i + 2)
   244→      if (endIdx < 0) {
   245→        tokens.push({ type: 'comment', value: line.slice(i) })
   246→        return { tokens, inBlockComment: true }
   247→      }
   248→      tokens.push({ type: 'comment', value: line.slice(i, endIdx + 2) })
   249→      i = endIdx + 2
   250→      continue
   251→    }
   252→
   253→    // String literals (single, double, backtick)
   254→    if (ch === '"' || ch === "'" || ch === '`') {
   255→      const quote = ch
   256→      let j = i + 1
   257→      while (j < line.length) {
   258→        if (line[j] === '\\') { j += 2; continue }
   259→        if (line[j] === quote) { j++; break }
   260→        j++
   261→      }
   262→      tokens.push({ type: 'string', value: line.slice(i, j) })
   263→      i = j
   264→      continue
   265→    }
   266→
   267→    // Numbers
   268→    if (/[0-9]/.test(ch) || (ch === '.' && /[0-9]/.test(line[i + 1] ?? ''))) {
   269→      let j = i + 1
   270→      while (j < line.length && /[0-9a-fA-FxXoObB._]/.test(line[j] ?? '')) j++
   271→      tokens.push({ type: 'number', value: line.slice(i, j) })
   272→      i = j
   273→      continue
   274→    }
   275→
   276→    // Identifiers and keywords
   277→    if (/[a-zA-Z_$]/.test(ch)) {
   278→      let j = i + 1
   279→      while (j < line.length && /[a-zA-Z0-9_$]/.test(line[j] ?? '')) j++
   280→      const word = line.slice(i, j)
   281→      // Check if followed by `(` → function call
   282→      let k = j
   283→      while (k < line.length && (line[k] === ' ' || line[k] === '\t')) k++
   284→      const isFunctionCall = line[k] === '('
   285→      if (keywords && keywords.has(word)) {
   286→        tokens.push({ type: 'keyword', value: word })
   287→      } else if (isFunctionCall) {
   288→        tokens.push({ type: 'function', value: word })
   289→      } else {
   290→        tokens.push({ type: 'plain', value: word })
   291→      }
   292→      i = j
   293→      continue
   294→    }
   295→
   296→    // Punctuation (single char)
   297→    tokens.push({ type: 'punctuation', value: ch })
   298→    i++
   299→  }
   300→
   301→  return { tokens, inBlockComment: blockComment }
   302→}

   304→/** Tokenize an HTML line. Recognizes tags, attributes, strings, and comments. */
   305→function tokenizeHtml(line: string, inBlockComment: boolean): { tokens: Token[]; inBlockComment: boolean } {
   306→  const tokens: Token[] = []
   307→  let i = 0
   308→  let inTag = false
   309→  let blockComment = inBlockComment
   310→
   311→  while (i < line.length) {
   312→    if (blockComment) {
   313→      const endIdx = line.indexOf('-->', i)
   314→      if (endIdx < 0) {
   315→        tokens.push({ type: 'comment', value: line.slice(i) })
   316→        return { tokens, inBlockComment: true }
   317→      }
   318→      tokens.push({ type: 'comment', value: line.slice(i, endIdx + 3) })
   319→      i = endIdx + 3
   320→      blockComment = false
   321→      continue
   322→    }
   323→
   324→    // HTML comment
   325→    if (line.startsWith('<!--', i)) {
   326→      const endIdx = line.indexOf('-->', i + 4)
   327→      if (endIdx < 0) {
   328→        tokens.push({ type: 'comment', value: line.slice(i) })
   329→        return { tokens, inBlockComment: true }
   330→      }
   331→      tokens.push({ type: 'comment', value: line.slice(i, endIdx + 3) })
   332→      i = endIdx + 3
   333→      continue
   334→    }
   335→
   336→    const ch = line[i]
   337→
   338→    // Tag start
   339→    if (ch === '<') {
   340→      // Find end of tag
   341→      const endIdx = line.indexOf('>', i)
   342→      if (endIdx < 0) {
   343→        // Malformed — push rest as plain
   344→        tokens.push({ type: 'plain', value: line.slice(i) })
   345→        i = line.length
   346→        continue
   347→      }
   348→      const tag = line.slice(i, endIdx + 1)
   349→      // Tokenize the tag content: <tagname attr="value" ...>
   350→      const tagTokens = tokenizeHtmlTag(tag)
   351→      tokens.push(...tagTokens)
   352→      i = endIdx + 1
   353→      inTag = false
   354→      continue
   355→    }
   356→
   357→    // Whitespace
   358→    if (ch === ' ' || ch === '\t') {
   359→      let j = i + 1
   360→      while (j < line.length && (line[j] === ' ' || line[j] === '\t')) j++
   361→      tokens.push({ type: 'plain', value: line.slice(i, j) })
   362→      i = j
   363→      continue
   364→    }
   365→
   366→    // Plain text content
   367→    let j = i
   368→    while (j < line.length && line[j] !== '<' && line[j] !== ' ' && line[j] !== '\t') j++
   369→    if (j === i) j++ // ensure progress
   370→    tokens.push({ type: 'plain', value: line.slice(i, j) })
   371→    i = j
   372→  }
   373→
   374→  return { tokens, inBlockComment: blockComment }
   375→}

   377→/** Tokenize a single HTML tag like `<div class="foo">` or `</div>`. */
   378→function tokenizeHtmlTag(tag: string): Token[] {
   379→  const tokens: Token[] = []
   380→  let i = 0
   381→  // Opening < (and optional /)
   382→  if (tag[i] === '<') {
   383→    tokens.push({ type: 'punctuation', value: '<' })
   384→    i++
   385→    if (tag[i] === '/') {
   386→      tokens.push({ type: 'punctuation', value: '/' })
   387→      i++
   388→    }
   389→  }
   390→  // Tag name
   391→  let j = i
   392→  while (j < tag.length && /[a-zA-Z0-9-]/.test(tag[j] ?? '')) j++
   393→  if (j > i) {
   394→    tokens.push({ type: 'tag', value: tag.slice(i, j) })
   395→    i = j
   396→  }
   397→  // Attributes
   398→  while (i < tag.length) {
   399→    const ch = tag[i]
   400→    // Skip whitespace
   401→    if (ch === ' ' || ch === '\t' || ch === '\n') {
   402→      let k = i + 1
   403→      while (k < tag.length && (tag[k] === ' ' || tag[k] === '\t' || tag[k] === '\n')) k++
   404→      tokens.push({ type: 'plain', value: tag.slice(i, k) })
   405→      i = k
   406→      continue
   407→    }
   408→    // End of tag
   409→    if (ch === '>' || (ch === '/' && tag[i + 1] === '>')) {
   410→      if (ch === '/') {
   411→        tokens.push({ type: 'punctuation', value: '/>' })
   412→        i += 2
   413→      } else {
   414→        tokens.push({ type: 'punctuation', value: '>' })
   415→        i++
   416→      }
   417→      continue
   418→    }
   419→    // Attribute name = value
   420→    let k = i
   421→    while (k < tag.length && /[a-zA-Z0-9-:]/.test(tag[k] ?? '')) k++
   422→    if (k > i) {
   423→      tokens.push({ type: 'attr', value: tag.slice(i, k) })
   424→      i = k
   425→      continue
   426→    }
   427→    // = sign
   428→    if (ch === '=') {
   429→      tokens.push({ type: 'punctuation', value: '=' })
   430→      i++
   431→      continue
   432→    }
   433→    // String value
   434→    if (ch === '"' || ch === "'") {
   435→      const quote = ch
   436→      let m = i + 1
   437→      while (m < tag.length && tag[m] !== quote) m++
   438→      if (m < tag.length) m++ // include closing quote
   439→      tokens.push({ type: 'string', value: tag.slice(i, m) })
   440→      i = m
   441→      continue
   442→    }
   443→    // Fallback: single char
   444→    tokens.push({ type: 'plain', value: ch })
   445→    i++
   446→  }
   447→  return tokens
   448→}

   450→/** Tokenize a CSS line. */
   451→function tokenizeCss(line: string, inBlockComment: boolean): { tokens: Token[]; inBlockComment: boolean } {
   452→  const tokens: Token[] = []
   453→  let i = 0
   454→  let blockComment = inBlockComment
   455→
   456→  while (i < line.length) {
   457→    if (blockComment) {
   458→      const endIdx = line.indexOf('*/', i)
   459→      if (endIdx < 0) {
   460→        tokens.push({ type: 'comment', value: line.slice(i) })
   461→        return { tokens, inBlockComment: true }
   462→      }
   463→      tokens.push({ type: 'comment', value: line.slice(i, endIdx + 2) })
   464→      i = endIdx + 2
   465→      blockComment = false
   466→      continue
   467→    }
   468→
   469→    const ch = line[i]
   470→
   471→    // Block comment
   472→    if (ch === '/' && line[i + 1] === '*') {
   473→      const endIdx = line.indexOf('*/', i + 2)
   474→      if (endIdx < 0) {
   475→        tokens.push({ type: 'comment', value: line.slice(i) })
   476→        return { tokens, inBlockComment: true }
   477→      }
   478→      tokens.push({ type: 'comment', value: line.slice(i, endIdx + 2) })
   479→      i = endIdx + 2
   480→      continue
   481→    }
   482→
   483→    // Strings
   484→    if (ch === '"' || ch === "'") {
   485→      const quote = ch
   486→      let j = i + 1
   487→      while (j < line.length && line[j] !== quote) {
   488→        if (line[j] === '\\') j += 2
   489→        else j++
   490→      }
   491→      if (j < line.length) j++
   492→      tokens.push({ type: 'string', value: line.slice(i, j) })
   493→      i = j
   494→      continue
   495→    }
   496→
   497→    // Numbers (with units like 12px, 1.5em, 100%)
   498→    if (/[0-9]/.test(ch) || (ch === '.' && /[0-9]/.test(line[i + 1] ?? ''))) {
   499→      let j = i + 1
   500→      while (j < line.length && /[0-9a-z%]/i.test(line[j] ?? '')) j++
   501→      tokens.push({ type: 'number', value: line.slice(i, j) })
   502→      i = j
   503→      continue
   504→    }
   505→
   506→    // Property/identifier: name followed by `:` is a property
   507→    if (/[a-zA-Z-]/.test(ch)) {
   508→      let j = i + 1
   509→      while (j < line.length && /[a-zA-Z0-9-_]/.test(line[j] ?? '')) j++
   510→      const word = line.slice(i, j)
   511→      // Look ahead for `:` → property name
   512→      let k = j
   513→      while (k < line.length && line[k] === ' ') k++
   514→      if (line[k] === ':') {
   515→        tokens.push({ type: 'attr', value: word })
   516→      } else {
   517→        tokens.push({ type: 'plain', value: word })
   518→      }
   519→      i = j
   520→      continue
   521→    }
   522→
   523→    // Hex colors (#abc123)
   524→    if (ch === '#' && /[0-9a-fA-F]/.test(line[i + 1] ?? '')) {
   525→      let j = i + 1
   526→      while (j < line.length && /[0-9a-fA-F]/.test(line[j] ?? '')) j++
   527→      tokens.push({ type: 'number', value: line.slice(i, j) })
   528→      i = j
   529→      continue
   530→    }
   531→
   532→    // Whitespace
   533→    if (ch === ' ' || ch === '\t') {
   534→      let j = i + 1
   535→      while (j < line.length && (line[j] === ' ' || line[j] === '\t')) j++
   536→      tokens.push({ type: 'plain', value: line.slice(i, j) })
   537→      i = j
   538→      continue
   539→    }
   540→
   541→    // Punctuation
   542→    tokens.push({ type: 'punctuation', value: ch })
   543→    i++
   544→  }
   545→
   546→  return { tokens, inBlockComment: blockComment }
   547→}

   549→/** Tokenize a JSON line. */
   550→function tokenizeJson(line: string): { tokens: Token[]; inBlockComment: boolean } {
   551→  const tokens: Token[] = []
   552→  let i = 0
   553→  while (i < line.length) {
   554→    const ch = line[i]
   555→    // Strings (also detect keys followed by `:`)
   556→    if (ch === '"') {
   557→      let j = i + 1
   558→      while (j < line.length) {
   559→        if (line[j] === '\\') { j += 2; continue }
   560→        if (line[j] === '"') { j++; break }
   561→        j++
   562→      }
   563→      // Look ahead for `:` (skipping whitespace) → it's a key
   564→      let k = j
   565→      while (k < line.length && (line[k] === ' ' || line[k] === '\t')) k++
   566→      const type = line[k] === ':' ? 'attr' : 'string'
   567→      tokens.push({ type, value: line.slice(i, j) })
   568→      i = j
   569→      continue
   570→    }
   571→    // Numbers (including negative and scientific)
   572→    if (/[0-9-]/.test(ch) && (/[0-9]/.test(line[i + 1] ?? '') || (ch === '-' && /[0-9]/.test(line[i + 1] ?? '')))) {
   573→      let j = i + 1
   574→      while (j < line.length && /[0-9eE+\-.]/.test(line[j] ?? '')) j++
   575→      tokens.push({ type: 'number', value: line.slice(i, j) })
   576→      i = j
   577→      continue
   578→    }
   579→    // Keywords: true, false, null
   580→    if (/[a-z]/.test(ch)) {
   581→      let j = i + 1
   582→      while (j < line.length && /[a-z]/.test(line[j] ?? '')) j++
   583→      const word = line.slice(i, j)
   584→      if (word === 'true' || word === 'false' || word === 'null') {
   585→        tokens.push({ type: 'keyword', value: word })
   586→      } else {
   587→        tokens.push({ type: 'plain', value: word })
   588→      }
   589→      i = j
   590→      continue
   591→    }
   592→    // Whitespace
   593→    if (ch === ' ' || ch === '\t') {
   594→      let j = i + 1
   595→      while (j < line.length && (line[j] === ' ' || line[j] === '\t')) j++
   596→      tokens.push({ type: 'plain', value: line.slice(i, j) })
   597→      i = j
   598→      continue
   599→    }
   600→    // Punctuation
   601→    tokens.push({ type: 'punctuation', value: ch })
   602→    i++
   603→  }
   604→  return { tokens, inBlockComment: false }
   605→}

   607→/** Tokenize a Markdown line. */
   608→function tokenizeMarkdown(line: string): Token[] {
   609→  const tokens: Token[] = []
   610→  // Heading
   611→  if (/^#{1,6}\s/.test(line)) {
   612→    const match = line.match(/^(#{1,6})(\s+.*)$/)
   613→    if (match) {
   614→      tokens.push({ type: 'keyword', value: match[1] })
   615→      tokens.push({ type: 'plain', value: match[2] })
   616→      return tokens
   617→    }
   618→  }
   619→  // List items
   620→  if (/^[-*+]\s/.test(line) || /^\d+\.\s/.test(line)) {
   621→    const match = line.match(/^([-*+]|\d+\.)\s(.*)$/)
   622→    if (match) {
   623→      tokens.push({ type: 'punctuation', value: match[1] })
   624→      tokens.push({ type: 'plain', value: ' ' })
   625→      tokens.push(...tokenizeMarkdownInline(match[2]))
   626→      return tokens
   627→    }
   628→  }
   629→  // Default: inline
   630→  return tokenizeMarkdownInline(line)
   631→}

   633→/** Tokenize inline markdown (bold, italic, code, links). */
   634→function tokenizeMarkdownInline(text: string): Token[] {
   635→  const tokens: Token[] = []
   636→  let i = 0
   637→  while (i < text.length) {
   638→    // Inline code: `code`
   639→    if (text[i] === '`') {
   640→      const endIdx = text.indexOf('`', i + 1)
   641→      if (endIdx > 0) {
   642→        tokens.push({ type: 'string', value: text.slice(i, endIdx + 1) })
   643→        i = endIdx + 1
   644→        continue
   645→      }
   646→    }
   647→    // Bold: **text**
   648→    if (text.startsWith('**', i)) {
   649→      const endIdx = text.indexOf('**', i + 2)
   650→      if (endIdx > 0) {
   651→        tokens.push({ type: 'keyword', value: text.slice(i, endIdx + 2) })
   652→        i = endIdx + 2
   653→        continue
   654→      }
   655→    }
   656→    // Italic: *text*
   657→    if (text[i] === '*') {
   658→      const endIdx = text.indexOf('*', i + 1)
   659→      if (endIdx > 0) {
   660→        tokens.push({ type: 'attr', value: text.slice(i, endIdx + 1) })
   661→        i = endIdx + 1
   662→        continue
   663→      }
   664→    }
   665→    // Plain text until next special char
   666→    let j = i + 1
   667→    while (j < text.length && text[j] !== '`' && text[j] !== '*') j++
   668→    tokens.push({ type: 'plain', value: text.slice(i, j) })
   669→    i = j
   670→  }
   671→  return tokens
   672→}

   688→// ── File tree builder ──
   689→
   690→interface TreeNode {
   691→  name: string
   692→  path: string
   693→  isFolder: boolean
   694→  children: TreeNode[]
   695→  file?: FileEntry
   696→}
   697→
   698→/**
   699→ * Build a tree from a flat list of file paths.
   700→ * Folders are created on-the-fly from path segments.
   701→ */
   702→function buildTree(files: FileEntry[]): TreeNode {
   703→  const root: TreeNode = { name: '', path: '', isFolder: true, children: [] }
   704→
   705→  for (const file of files) {
   706→    const parts = file.path.split('/').filter(Boolean)
   707→    let node = root
   708→    for (let i = 0; i < parts.length; i++) {
   709→      const part = parts[i]!
   710→      const isLeaf = i === parts.length - 1
   711→      const fullPath = parts.slice(0, i + 1).join('/')
   712→      let child = node.children.find(c => c.name === part)
   713→      if (!child) {
   714→        child = {
   715→          name: part,
   716→          path: fullPath,
   717→          isFolder: !isLeaf,
   718→          children: [],
   719→          file: isLeaf ? file : undefined,
   720→        }
   721→        node.children.push(child)
   722→      }
   723→      node = child
   724→    }
   725→  }
   726→
   727→  // Sort: folders first, then files, alphabetically
   728→  const sortNode = (n: TreeNode): void => {
   729→    n.children.sort((a, b) => {
   730→      if (a.isFolder !== b.isFolder) return a.isFolder ? -1 : 1
   731→      return a.name.localeCompare(b.name)
   732→    })
   733→    n.children.forEach(sortNode)
   734→  }
   735→  sortNode(root)
   736→
   737→  return root
   738→}

   742→export function FileViewer({ files, title = 'Files', className = '' }: FileViewerProps) {
   743→  const [selectedPath, setSelectedPath] = useState<string | null>(
   744→    files[0]?.path ?? null,
   745→  )
   746→  const [collapsedFolders, setCollapsedFolders] = useState<Set<string>>(new Set())
   747→  const [copiedPath, setCopiedPath] = useState<string | null>(null)
   748→
   749→  const tree = useMemo(() => buildTree(files), [files])
   750→
   751→  const selectedFile = useMemo(
   752→    () => files.find(f => f.path === selectedPath) ?? files[0] ?? null,
   753→    [files, selectedPath],
   754→  )
   755→
   756→  const toggleFolder = useCallback((path: string) => {
   757→    setCollapsedFolders(prev => {
   758→      const next = new Set(prev)
   759→      if (next.has(path)) next.delete(path)
   760→      else next.add(path)
   761→      return next
   762→    })
   763→  }, [])
   764→

   766→    try {
   767→      await navigator.clipboard.writeText(file.content)
   768→      setCopiedPath(file.path)
   769→      setTimeout(() => setCopiedPath(null), 1500)
   770→    } catch {
   771→      // Clipboard may be unavailable (SSR, insecure context). Fail silently.
   772→    }
   773→  }, [])
   774→
   775→  const handleDownloadFile = useCallback((file: FileEntry) => {
   776→    try {
   777→      const blob = new Blob([file.content], { type: 'text/plain;charset=utf-8' })
   778→      const url = URL.createObjectURL(blob)
   779→      const a = document.createElement('a')
   780→      a.href = url
   781→      a.download = file.path.split('/').pop() ?? 'file.txt'
   782→      document.body.appendChild(a)
   783→      a.click()
   784→      document.body.removeChild(a)
   785→      URL.revokeObjectURL(url)
   786→    } catch {
   787→      // Fail silently — download is best-effort
   788→    }
   789→  }, [])
   790→
   791→  const handleDownloadZip = useCallback(() => {
   792→    try {
   793→      const zipBytes = createZip(
   794→        files.map(f => ({ name: f.path, content: f.content })),
   795→      )
   796→      const blob = new Blob([zipBytes as BlobPart], { type: 'application/zip' })
   797→      const url = URL.createObjectURL(blob)
   798→      const a = document.createElement('a')
   799→      a.href = url
   800→      a.download = 'nova-project.zip'
   801→      document.body.appendChild(a)
   802→      a.click()
   803→      document.body.removeChild(a)
   804→      URL.revokeObjectURL(url)
   805→    } catch {
   806→      // Fail silently
   807→    }
   808→  }, [files])
   809→
   810→  if (!files || files.length === 0) {
   811→    return (
   812→      <div className={`flex items-center justify-center p-8 text-sm text-muted-foreground ${className}`}>
   813→        No files to display
   814→      </div>
   815→    )
   816→  }
   817→
   818→  return (
   819→    <div className={`flex h-full min-h-[300px] flex-col overflow-hidden rounded-md border border-border/40 bg-card/30 ${className}`}>
   820→      {/* Header */}
   821→      <div className="flex items-center justify-between border-b border-border/40 bg-card/60 px-3 py-2">
   822→        <span className="text-xs font-medium text-foreground">{title}</span>

   824→          type="button"
   825→          onClick={handleDownloadZip}
   826→          className="flex items-center gap-1 rounded px-2 py-1 text-[11px] text-muted-foreground transition-colors hover:bg-muted/40 hover:text-foreground"
   827→          title="Download all as ZIP"
   828→        >
   829→          <FileArchive className="h-3 w-3" />
   830→          ZIP
   831→        </button>
   832→      </div>

   838→          <TreeView
   839→            node={tree}
   840→            level={0}
   841→            selectedPath={selectedPath}
   842→            collapsedFolders={collapsedFolders}
   843→            onSelect={setSelectedPath}
   844→            onToggle={toggleFolder}
   845→          />
   846→        </div>
   847→
   848→        {/* File content */}

   850→          <div className="flex min-h-0 min-w-0 flex-1 flex-col">
   851→            <FileHeader

   853→              copied={copiedPath === selectedFile.path}
