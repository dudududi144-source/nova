    88→  // Save history to localStorage — pure side effect, called OUTSIDE of setState updaters
    89→  // to avoid double-firing in React StrictMode.
    90→  const saveHistoryToStorage = useCallback((items: BuildResult[]) => {
    91→    let savedCount = items.length
    92→    try {
    93→      localStorage.setItem('nova_history', JSON.stringify(items))
    94→    } catch (quotaErr) {
    95→      console.error('[NOVA] localStorage quota exceeded:', quotaErr)
    96→      savedCount = 0
    97→      for (let i = items.length - 1; i >= 0; i--) {
    98→        try {
    99→          localStorage.setItem('nova_history', JSON.stringify(items.slice(0, i + 1)))
   100→          savedCount = i + 1
   101→          break
   102→        } catch {
   103→          // keep trying smaller
   104→        }
   105→      }
   106→    }
   107→    if (savedCount < items.length) {
   108→      toast.error(`localStorage full — only ${savedCount} of ${items.length} builds saved to history`)
   109→    }
   110→  }, [])

   112→  // Add a build result to history (dedupe by mission) and persist to localStorage.
   113→  // Side effects (localStorage, toast) happen here — NOT inside the setState updater.
   114→  const addBuildToHistory = useCallback((buildResult: BuildResult) => {

   118→      return newHistory

   124→  // Abort any in-flight build AND refine on unmount

   129→    }

   131→
   132→  // Elapsed time counter + dynamic thinking step rotation
   133→  useEffect(() => {
   134→    if (!loading && !refining) {
   135→      setElapsed(0)
   136→      setThinkingStep(0)
   137→      return
   138→    }
   139→    const startTime = Date.now()
   140→    const steps = loading ? buildSteps : REFINE_THINKING_STEPS
   141→    let step = 0
   142→    setThinkingStep(0)
   143→
   144→    const timer = setInterval(() => {
   145→      const sec = Math.floor((Date.now() - startTime) / 1000)
   146→      setElapsed(sec)
   147→
   148→      // Rotate through steps — timing depends on how many steps we have
   149→      // Allocate ~4 seconds per step, but always keep the last step if we run out
   150→      const stepDuration = loading ? 4 : 5 // 4s for build (more steps), 5s for refine
   151→      const nextStep = Math.min(steps.length - 1, Math.floor(sec / stepDuration))
   152→      if (nextStep !== step) {
   153→        step = nextStep
   154→        setThinkingStep(step)
   155→      }
   156→    }, 1000)
   157→    return () => clearInterval(timer)
   158→  }, [loading, refining, buildSteps])
   159→
   160→  // Centralized build function. Aborts any in-flight build first.
   161→  // Accepts an optional explicit mission to avoid stale-closure bugs (e.g., retry).
   162→  // Uses resultRef instead of result state to avoid being re-created on every build.
   163→  const build = useCallback(async (explicitMission?: string) => {
   164→    const m = (explicitMission ?? mission).trim()
   165→    if (!m) {
   166→      toast.error('Describe what to build first')
   167→      return
   168→    }
   169→
   170→    // If an explicit mission was passed, sync the textarea state
   171→    if (explicitMission && explicitMission !== mission) {
   172→      setMission(explicitMission)
   173→    }
   174→
   175→    // Abort any in-flight build (covers: rebuild, history-click-during-build, reset-during-build)
   176→    abortRef.current?.abort()
   177→    const controller = new AbortController()
   178→    abortRef.current = controller
   179→
   180→    setLoading(true)
   181→    setError(null)
   182→    setFailedMission(null)
   183→    setChatMessages([])
   184→
   185→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   186→    const steps = extractStepsFromMission(m)
   187→    setBuildSteps(steps)
   188→    setPlanSummary(null)
   189→    setLivePreviewHtml(null)
   190→
   191→    // Helper: set error state consistently (replaces 6 repeated blocks)
   192→    const fail = (msg: string) => {
   193→      setError(msg)
   194→      setFailedMission(m)
   195→      if (!resultRef.current) toast.error(msg)
   196→    }
   197→
   198→    try {
   199→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   200→      // This is a separate, fast call (~2-3s). The plan lets us show
   201→      // AUTHENTIC steps to the user before the coder even starts.
   202→      const archRes = await fetch('/api/build/architect', {
   203→        method: 'POST',
   204→        headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify({ mission: m }),
   206→        signal: controller.signal,
   207→      })
   208→
   209→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   210→      const archContentType = archRes.headers.get('content-type') ?? ''
   211→      if (archContentType.includes('application/json')) {
   212→        try { archData = await archRes.json() } catch {}
   213→      }
   214→
   215→      // If architect succeeded, update steps with the REAL plan
   216→      if (archData?.ok && archData.plan) {
   217→        const planSteps = extractStepsFromPlan(archData.plan, m)
   218→        const summary = getPlanSummary(archData.plan)
   219→        setBuildSteps(planSteps)
   220→        setPlanSummary(summary)
   221→        console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   222→      }
   223→      // If architect failed, continue with mission-based steps (already set)
   224→
   225→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   226→      // The route returns Server-Sent Events: progress events while LLM works,
   227→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   228→      const codeRes = await fetch('/api/build/code', {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null }),
   232→        signal: controller.signal,
   233→      })
   234→
   235→      if (!codeRes.ok) {
   236→        // Non-SSE error (400, 429, 413) — parse as JSON
   237→        let errorMsg = `Server error (${codeRes.status})`
   238→        try {
   239→          const errData = await codeRes.json()
   240→          if (errData?.error) errorMsg = errData.error
   241→        } catch {}
   242→        fail(errorMsg)
   243→        return
   244→      }
   245→
   246→      // Read SSE stream
   247→      const reader = codeRes.body?.getReader()
   248→      if (!reader) {
   249→        fail('No response stream')
   250→        return
   251→      }
   252→
   253→      const decoder = new TextDecoder()
   254→      let buffer = ''
   255→      let finalHtml = ''
   256→      let finalTokens = 0
   257→      let finalMs = 0
   258→      let finalQuality = 0
   259→      let finalMetrics = ''
   260→      let streamError: string | null = null
   261→
   262→      while (true) {
   263→        const { done, value } = await reader.read()
   264→        if (done) break
   265→
   266→        buffer += decoder.decode(value, { stream: true })
   267→
   268→        // Parse SSE events (separated by \n\n)
   269→        const events = buffer.split('\n\n')
   270→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   271→
   272→        for (const eventStr of events) {
   273→          const dataLine = eventStr.trim()
   274→          if (!dataLine.startsWith('data: ')) continue
   275→
   276→          try {
   277→            const evt = JSON.parse(dataLine.slice(6))
   278→
   279→            if (evt.type === 'progress') {
   280→              // Update thinking steps with real progress from server
   281→              setBuildSteps(prev => {
   282→                // Replace with server-provided step, keep last as fallback
   283→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   284→                return [prev[0], evt.step, ...prev.slice(2)]
   285→              })
   286→            } else if (evt.type === 'token') {
   287→              // REAL TOKEN STREAMING — accumulate partial HTML for live preview
   288→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   289→              setBuildSteps(prev => {
   290→                const last = prev[prev.length - 1]
   291→                const totalLen = (evt.length ?? 0)
   292→                const newLast = `Generating: ${totalLen} chars...`
   293→                if (last && last.startsWith('Generating:')) {
   294→                  return [...prev.slice(0, -1), newLast]
   295→                }
   296→                return [...prev, newLast]
   297→              })
   298→            } else if (evt.type === 'result') {
   299→              finalHtml = evt.html ?? ''
   300→              finalTokens = evt.tokens ?? 0
   301→              finalMs = evt.ms ?? 0
   302→              finalQuality = evt.quality ?? 0
   303→              finalMetrics = evt.metrics ?? ''
   304→            } else if (evt.type === 'error') {
   305→              streamError = evt.error ?? 'Unknown error'
   306→            }
   307→          } catch {
   308→            // Ignore malformed events
   309→          }
   310→        }
   311→      }
   312→
   313→      if (streamError) {
   314→        fail(streamError)
   315→        return
   316→      }
   317→
   318→      if (!finalHtml) {
   319→        fail('Server returned empty HTML')
   320→        return
   321→      }
   322→
   323→      const buildResult: BuildResult = {
   324→        id: newBuildId(),
   325→        html: finalHtml,
   326→        tokens: finalTokens,
   327→        ms: finalMs,
   328→        mission: m,
   329→      }
   330→
   331→      setResult(buildResult)
   332→      addBuildToHistory(buildResult)
   333→
   334→      toast.success(`Built in ${(finalMs / 1000).toFixed(1)}s · ${finalTokens} tokens · quality: ${finalQuality}`)
   335→      setQualityScore(finalQuality)
   336→      setQualityMetrics(finalMetrics)
   337→    } catch (err: unknown) {
   338→      // AbortError = user started a new build, loaded history, or navigated away; silently ignore
   339→      if (err instanceof DOMException && err.name === 'AbortError') return
   340→      const msg = err instanceof Error ? err.message : 'Network error'
   341→      fail(msg)
   342→    } finally {
   343→      // Only clear loading if this controller is still the active one
   344→      if (abortRef.current === controller) {
   345→        abortRef.current = null
   346→        setLoading(false)
   347→      }
   348→    }
   349→  }, [mission])
   350→
   351→  const loadFromHistory = useCallback((h: BuildResult) => {
   352→    // Abort any in-flight build or refine
   353→    abortRef.current?.abort()
   354→    abortRef.current = null
   355→    refineAbortRef.current?.abort()
   356→    refineAbortRef.current = null
   357→    setLoading(false)
   358→    setRefining(false)
   359→    setResult(h)
   360→    setMission(h.mission)
   361→    setError(null)
   362→    setFailedMission(null)
   363→    setChatMessages([])
   364→    // Reset all derived state so we don't show the previous build's badges/plan
   365→    setQualityScore(0)
   366→    setQualityMetrics('')
   367→    setPlanSummary(null)
   368→    setLivePreviewHtml(null)
   369→    setConfirmClear(false)
   370→  }, [])
   371→
   372→  const cancelBuild = useCallback(() => {
   373→    // Cancel only aborts the in-flight build — does NOT clear mission or result.
   374→    // The user keeps their mission text and the old preview (if any).
   375→    abortRef.current?.abort()
   376→    abortRef.current = null
   377→    setLoading(false)
   378→    setError(null)
   379→    setFailedMission(null)
   380→  }, [])
   381→
   382→  const reset = useCallback(() => {
   383→    // Reset clears everything — used by the "New" button.
   384→    // Abort BOTH in-flight build and refine to prevent phantom state mutations.
   385→    abortRef.current?.abort()
   386→    abortRef.current = null
   387→    refineAbortRef.current?.abort()
   388→    refineAbortRef.current = null
   389→    setLoading(false)
   390→    setRefining(false)
   391→    setResult(null)
   392→    setError(null)
   393→    setFailedMission(null)
   394→    setMission('')
   395→    setChatMessages([])
   396→    setQualityScore(0)
   397→    setQualityMetrics('')
   398→    setLivePreviewHtml(null)
   399→    setPlanSummary(null)
   400→    setConfirmClear(false)
   401→  }, [])
   402→
   403→  const retryFailed = useCallback(() => {
   404→    // If the user edited the textarea after the failure, use the edited mission.
   405→    // Otherwise, use the original failed mission.
   406→    const currentMission = mission.trim()
   407→    if (currentMission && currentMission !== failedMission) {
   408→      build(currentMission)
   409→    } else if (failedMission) {
   410→      build(failedMission)
   411→    }
   412→  }, [failedMission, build, mission])
   413→
   414→  const download = useCallback(() => {
   415→    if (!result?.html) return
   416→    const blob = new Blob([result.html], { type: 'text/html' })
   417→    const url = URL.createObjectURL(blob)
   418→    const a = document.createElement('a')
   419→    a.href = url
   420→    const filename = sanitizeFilename(result.mission)
   421→    a.download = filename
   422→    document.body.appendChild(a)
   423→    a.click()
   424→    document.body.removeChild(a)
   425→    URL.revokeObjectURL(url)
   426→    toast.success(`Downloaded ${filename}`)
   427→  }, [result])
   428→
   429→  // Copy HTML to clipboard
   430→  const copyHtml = useCallback(async () => {
   431→    if (!result?.html) return
   432→    try {
   433→      await navigator.clipboard.writeText(result.html)
   434→      toast.success('HTML copied to clipboard')

   436→      toast.error('Failed to copy — try downloading instead')
   437→    }
   438→  }, [result])
   439→
   440→  // Open the HTML in a new browser tab (full-screen preview)
   441→  const openInNewTab = useCallback(() => {
   442→    if (!result?.html) return
   443→    const blob = new Blob([result.html], { type: 'text/html' })
   444→    const url = URL.createObjectURL(blob)
   445→    window.open(url, '_blank', 'noopener,noreferrer')
   446→    // Revoke after 30s — enough time for the tab to load
   447→    setTimeout(() => URL.revokeObjectURL(url), 30_000)
   448→    toast.info('Opened in new tab')
   449→  }, [result])
   450→
   451→  // Chat refine: SSE streaming — same pattern as build/code
   452→  const sendChat = useCallback(async () => {
   453→    const msg = chatInput.trim()
   454→    const currentResult = resultRef.current
   455→    if (!msg || refining) return
   456→    if (!currentResult) {
   457→      toast.info('Build something first, then you can refine it')
   458→      return
   459→    }
   460→
   461→    const userMsg: ChatMessage = { role: 'user', content: msg, ts: Date.now() }
   462→    setChatMessages(prev => [...prev, userMsg])
   463→    setChatInput('')
   464→    setRefining(true)
   465→
   466→    refineAbortRef.current?.abort()
   467→    const controller = new AbortController()
   468→    refineAbortRef.current = controller
   469→
   470→    // Clear live preview so refine starts fresh
   471→    setLivePreviewHtml(null)
   472→
   473→    try {
   474→      const res = await fetch('/api/refine', {
   475→        method: 'POST',
   476→        headers: { 'Content-Type': 'application/json' },
   477→        body: JSON.stringify({ mission: currentResult.mission, html: currentResult.html, message: msg }),
   478→        signal: controller.signal,
   479→      })
   480→
   481→      if (!res.ok) {
   482→        let errorMsg = `Server error (${res.status})`
   483→        try { const e = await res.json(); if (e?.error) errorMsg = e.error } catch {}
   484→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${errorMsg}`, ts: Date.now() }])
   485→        toast.error(errorMsg)
   486→        return
   487→      }
   488→
   489→      // Read SSE stream
   490→      const reader = res.body?.getReader()
   491→      if (!reader) {
   492→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: No stream', ts: Date.now() }])
   493→        return
   494→      }
   495→
   496→      const decoder = new TextDecoder()
   497→      let buffer = ''
   498→      let finalHtml = ''
   499→      let finalTokens = 0
   500→      let finalMs = 0
   501→      let finalQuality = 0
   502→      let finalMetrics = ''
   503→      let streamError: string | null = null
   504→
   505→      while (true) {
   506→        const { done, value } = await reader.read()
   507→        if (done) break
   508→
   509→        buffer += decoder.decode(value, { stream: true })
   510→        const events = buffer.split('\n\n')
   511→        buffer = events.pop() ?? ''
   512→
   513→        for (const eventStr of events) {
   514→          const dataLine = eventStr.trim()
   515→          if (!dataLine.startsWith('data: ')) continue
   516→          try {
   517→            const evt = JSON.parse(dataLine.slice(6))
   518→            if (evt.type === 'token') {
   519→              // Live token streaming — show refined HTML appearing in real-time
   520→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   521→            } else if (evt.type === 'result') {
   522→              finalHtml = evt.html ?? ''
   523→              finalTokens = evt.tokens ?? 0
   524→              finalMs = evt.ms ?? 0
   525→              finalQuality = evt.quality ?? 0
   526→              finalMetrics = evt.metrics ?? ''
   527→            } else if (evt.type === 'error') {
   528→              streamError = evt.error ?? 'Unknown error'
   529→            }
   530→          } catch {}
   531→        }
   532→      }
   533→
   534→      if (streamError) {
   535→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${streamError}`, ts: Date.now() }])
   536→        toast.error(streamError)
   537→        return
   538→      }
   539→
   540→      if (!finalHtml) {
   541→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: Empty response', ts: Date.now() }])
   542→        return
   543→      }
   544→
   545→      const refinedResult: BuildResult = {
   546→        ...currentResult,
   547→        id: newBuildId(),
   548→        html: finalHtml,
   549→      }
   550→      setResult(refinedResult)
   551→      addBuildToHistory(refinedResult)
   552→
   553→      setChatMessages(prev => [...prev, {
   554→        role: 'assistant',
   555→        content: `Updated! ${finalMs ? `${(finalMs / 1000).toFixed(1)}s` : ''} · ${finalTokens} tokens`,
   556→        ts: Date.now(),
   557→      }])
   558→
   559→      toast.success(`Refined! ${(finalMs / 1000).toFixed(1)}s · quality: ${finalQuality}`)
   560→      setQualityScore(finalQuality)
   561→      setQualityMetrics(finalMetrics)
   562→    } catch (err: unknown) {
   563→      if (err instanceof DOMException && err.name === 'AbortError') return
   564→      const failMsg = err instanceof Error ? err.message : 'Network error'
