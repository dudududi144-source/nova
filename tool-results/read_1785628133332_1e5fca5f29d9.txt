   371→   371→        headers: { 'Content-Type': 'application/json' },
   372→   372→        body: JSON.stringify({ mission: m }),
   373→   373→        signal: controller.signal,
   374→   374→      })
   375→   375→
   376→   376→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   377→   377→      const archContentType = archRes.headers.get('content-type') ?? ''
   378→   378→      if (archContentType.includes('application/json')) {
   379→   379→        try { archData = await archRes.json() } catch {}
   380→   380→      }
   381→   381→
   382→   382→      // If architect succeeded, update steps with the REAL plan
   383→   383→      if (archData?.ok && archData.plan) {

   385→   385→        const summary = getPlanSummary(archData.plan)
   386→   386→        setBuildSteps(planSteps)
   387→   387→        setPlanSummary(summary)

   389→   389→        if (process.env.NODE_ENV !== 'production') {

   391→   391→        }
   392→   392→      }
   393→   393→      // If architect failed, continue with mission-based steps (already set)
   394→   394→
   395→   395→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   396→   396→      // The route returns Server-Sent Events: progress events while LLM works,
   397→   397→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   398→   398→      const codeRes = await fetch('/api/build/code', {
   399→   399→        method: 'POST',
   400→   400→        headers: { 'Content-Type': 'application/json', 'Accept': 'text/event-stream' },
   401→   401→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null, theme: selectedTheme }),
   402→   402→        signal: controller.signal,
   403→   403→      })
   404→   404→
   405→   405→      if (!codeRes.ok) {
   406→   406→        // Non-SSE error (400, 429, 413) — parse as JSON
   407→   407→        let errorMsg = `Server error (${codeRes.status})`
   408→   408→        try {
   409→   409→          const errData = await codeRes.json()
   410→   410→          if (errData?.error) errorMsg = errData.error
   411→   411→        } catch {}
   412→   412→        fail(errorMsg)
   413→   413→        return
   414→   414→      }
   415→   415→
   416→   416→      // Verify the response is actually SSE — a proxy or CDN might return 200 with HTML
   417→   417→      // (captive portal, error page) which would silently fail to parse as SSE.

   419→   419→      if (!codeCt.includes('text/event-stream')) {
   420→   420→        let errorMsg = 'Unexpected response from server'
   421→   421→        try {
   422→   422→          const errData = await codeRes.json()
   423→   423→          if (errData?.error) errorMsg = errData.error
   424→   424→        } catch {}
   425→   425→        fail(errorMsg)
   426→   426→        return
   427→   427→      }
   428→   428→
   429→   429→      // Read SSE stream
   430→   430→      const reader = codeRes.body?.getReader()
   431→   431→      if (!reader) {
   432→   432→        fail('No response stream')
   433→   433→        return
