   275→  // Elapsed time counter + dynamic thinking step rotation.

   287→    setThinkingStep(0)
   288→

   290→      const sec = Math.floor((Date.now() - startTime) / 1000)

   292→

   296→
   297→      // Rotate through steps — ~4s per step for build, 5s for refine
   298→      const stepDuration = loading ? 4 : 5
   299→      const nextStep = Math.min(steps.length - 1, Math.floor(sec / stepDuration))
   300→      if (nextStep !== step) {
   301→        step = nextStep
   302→        setThinkingStep(step)
   303→      }
   304→    }, 1000)
   305→    return () => clearInterval(timer)
   306→  }, [loading, refining])

   308→  // Centralized build function. Aborts any in-flight build first.
   309→  // Accepts an optional explicit mission to avoid stale-closure bugs (e.g., retry).
   310→  // Uses resultRef instead of result state to avoid being re-created on every build.
   311→  const build = useCallback(async (explicitMission?: string) => {
   312→    const m = (explicitMission ?? mission).trim()
   313→    if (!m) {
   314→      toast.error('Describe what to build first')
   315→      return
   316→    }
   317→
   318→    // If an explicit mission was passed, sync the textarea state
   319→    if (explicitMission && explicitMission !== mission) {
   320→      setMission(explicitMission)
   321→    }
   322→
   323→    // Abort any in-flight build AND refine (covers: rebuild during refine, history-click-during-build, reset-during-build)
   324→    // This prevents the race condition where build and refine run simultaneously and
   325→    // both call setResult/setHistory, corrupting state.
   326→    abortRef.current?.abort()
   327→    refineAbortRef.current?.abort()
   328→    refineAbortRef.current = null
   329→    setRefining(false)
   330→    const controller = new AbortController()
   331→    abortRef.current = controller
   332→
   333→    setLoading(true)
   334→    setError(null)
   335→    setFailedMission(null)
   336→    setChatMessages([])
   337→    setConfirmClear(false) // Hide confirm-clear UI if it was visible
   338→
   339→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   340→    const steps = extractStepsFromMission(m)
   341→    setBuildSteps(steps)
   342→    setPlanSummary(null)
   343→    setLivePreviewHtml(null)
   344→    livePreviewAccumulatorRef.current = '' // Clear accumulator for fresh build
   345→    generatingCharsRef.current = 0 // Reset char counter for build step throttling
   346→
   347→    // Helper: set error state consistently (replaces 6 repeated blocks)
   348→    const fail = (msg: string) => {
   349→      setError(msg)
   350→      setFailedMission(m)
   351→      if (!resultRef.current) toast.error(msg)
   352→    }
   353→
   354→    try {
   355→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   356→      // This is a separate, fast call (~2-3s). The plan lets us show
   357→      // AUTHENTIC steps to the user before the coder even starts.
   358→      const archRes = await fetch('/api/build/architect', {
   359→        method: 'POST',
   360→        headers: { 'Content-Type': 'application/json' },
   361→        body: JSON.stringify({ mission: m }),
   362→        signal: controller.signal,
   363→      })
   364→
   365→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   366→      const archContentType = archRes.headers.get('content-type') ?? ''
   367→      if (archContentType.includes('application/json')) {
   368→        try { archData = await archRes.json() } catch {}
   369→      }
   370→
   371→      // If architect succeeded, update steps with the REAL plan
   372→      if (archData?.ok && archData.plan) {
   373→        const planSteps = extractStepsFromPlan(archData.plan, m)
   374→        const summary = getPlanSummary(archData.plan)
   375→        setBuildSteps(planSteps)
   376→        setPlanSummary(summary)
   377→        // Debug log gated to non-production — don't spam user's console
   378→        if (process.env.NODE_ENV !== 'production') {
   379→          console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   380→        }
   381→      }
   382→      // If architect failed, continue with mission-based steps (already set)
   383→
   384→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   385→      // The route returns Server-Sent Events: progress events while LLM works,
   386→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   387→      const codeRes = await fetch('/api/build/code', {
   388→        method: 'POST',
   389→        headers: { 'Content-Type': 'application/json', 'Accept': 'text/event-stream' },
   390→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null, theme: selectedTheme }),
   391→        signal: controller.signal,
   392→      })
   393→
   394→      if (!codeRes.ok) {
   395→        // Non-SSE error (400, 429, 413) — parse as JSON
   396→        let errorMsg = `Server error (${codeRes.status})`
   397→        try {
   398→          const errData = await codeRes.json()
   399→          if (errData?.error) errorMsg = errData.error
   400→        } catch {}
   401→        fail(errorMsg)
   402→        return
   403→      }
   404→
   405→      // Verify the response is actually SSE — a proxy or CDN might return 200 with HTML
   406→      // (captive portal, error page) which would silently fail to parse as SSE.
   407→      const codeCt = codeRes.headers.get('content-type') ?? ''
   408→      if (!codeCt.includes('text/event-stream')) {
   409→        let errorMsg = 'Unexpected response from server'
   410→        try {
   411→          const errData = await codeRes.json()
   412→          if (errData?.error) errorMsg = errData.error
   413→        } catch {}
   414→        fail(errorMsg)
   415→        return
   416→      }
   417→
   418→      // Read SSE stream
   419→      const reader = codeRes.body?.getReader()
   420→      if (!reader) {
   421→        fail('No response stream')
   422→        return
   423→      }
   424→
   425→      const decoder = new TextDecoder()
   426→      let buffer = ''
   427→      let finalHtml = ''
   428→      let finalTokens = 0
   429→      let finalMs = 0
   430→      let finalQuality = 0
   431→      let finalMetrics = ''
   432→      let streamError: string | null = null
   433→
   434→      while (true) {
   435→        const { done, value } = await reader.read()
   436→        if (done) break
   437→
   438→        buffer += decoder.decode(value, { stream: true })
   439→
   440→        // Parse SSE events (separated by \n\n or \r\n\r\n — some proxies normalize to \r\n)
   441→        // Normalize \r\n to \n first so the split works regardless of proxy behavior.
   442→        const normalized = buffer.replace(/\r\n/g, '\n')
   443→        const events = normalized.split('\n\n')
   444→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   445→
   446→        for (const eventStr of events) {
   447→          const dataLine = eventStr.trim()
   448→          if (!dataLine.startsWith('data: ')) continue
   449→
   450→          try {
   451→            const evt = JSON.parse(dataLine.slice(6))
   452→
   453→            if (evt.type === 'progress') {
   454→              // Update thinking steps with real progress from server
   455→              setBuildSteps(prev => {
   456→                // Replace with server-provided step, keep last as fallback
   457→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   458→                return [prev[0], evt.step, ...prev.slice(2)]
   459→              })
   460→            } else if (evt.type === 'token') {
   461→              // REAL TOKEN STREAMING — accumulate in ref for throttled flush.
   462→              // We DON'T call setLivePreviewHtml here (would reload iframe on every token).
   463→              // The 200ms flush interval picks up the accumulated text.
   464→              livePreviewAccumulatorRef.current = (livePreviewAccumulatorRef.current || '') + (evt.text ?? '')
   465→              // Throttle setBuildSteps: only update the "Generating: N chars..." text
   466→              // when the char count crosses a 500-char threshold (not on every token).
   467→              // This prevents ~2000 setBuildSteps calls (each creating a new array) per build.
   468→              const totalLen = (evt.length ?? 0)
   469→              const lastUpdate = generatingCharsRef.current
   470→              if (totalLen - lastUpdate >= 500 || totalLen === 0) {
   471→                generatingCharsRef.current = totalLen
   472→                setBuildSteps(prev => {
   473→                  const last = prev[prev.length - 1]
   474→                  const newLast = `Generating: ${totalLen} chars...`
   475→                  if (last && last.startsWith('Generating:')) {
   476→                    return [...prev.slice(0, -1), newLast]
   477→                  }
   478→                  return [...prev, newLast]
   479→                })
   480→              }
   481→            } else if (evt.type === 'result') {
   482→              finalHtml = evt.html ?? ''
   483→              finalTokens = evt.tokens ?? 0
   484→              finalMs = evt.ms ?? 0
   485→              finalQuality = evt.quality ?? 0
   486→              finalMetrics = evt.metrics ?? ''
   487→            } else if (evt.type === 'error') {
   488→              streamError = evt.error ?? 'Unknown error'
   489→            }
   490→          } catch {
   491→            // Ignore malformed events
   492→          }
   493→        }
   494→      }
   495→
   496→      // Flush the decoder — any remaining bytes (incomplete multi-byte chars)
   497→      buffer += decoder.decode()
   498→      // Process any remaining complete event in the buffer
   499→      if (buffer.trim()) {
   500→        const normalized = buffer.replace(/\r\n/g, '\n')
   501→        const events = normalized.split('\n\n')
   502→        for (const eventStr of events) {
   503→          const dataLine = eventStr.trim()
   504→          if (!dataLine.startsWith('data: ')) continue
   505→          try {
   506→            const evt = JSON.parse(dataLine.slice(6))
   507→            if (evt.type === 'result') {
   508→              finalHtml = evt.html ?? ''
   509→              finalTokens = evt.tokens ?? 0
   510→              finalMs = evt.ms ?? 0
   511→              finalQuality = evt.quality ?? 0
   512→              finalMetrics = evt.metrics ?? ''
   513→            } else if (evt.type === 'error') {
   514→              streamError = evt.error ?? 'Unknown error'
   515→            }
   516→          } catch {}
   517→        }
   518→      }
   519→
   520→      if (streamError) {
   521→        fail(streamError)
   522→        return
