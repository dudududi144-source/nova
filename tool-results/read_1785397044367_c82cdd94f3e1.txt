   105→      const nextStep = Math.min(steps.length - 1, Math.floor(sec / stepDuration))

   107→        step = nextStep

   114→  // Centralized build function. Aborts any in-flight build first.
   115→  // Accepts an optional explicit mission to avoid stale-closure bugs (e.g., retry).
   116→  // Uses resultRef instead of result state to avoid being re-created on every build.
   117→  const build = useCallback(async (explicitMission?: string) => {
   118→    const m = (explicitMission ?? mission).trim()
   119→    if (!m) {
   120→      toast.error('Describe what to build first')
   121→      return
   122→    }
   123→
   124→    // If an explicit mission was passed, sync the textarea state
   125→    if (explicitMission && explicitMission !== mission) {
   126→      setMission(explicitMission)
   127→    }
   128→
   129→    // Abort any in-flight build (covers: rebuild, history-click-during-build, reset-during-build)
   130→    abortRef.current?.abort()
   131→    const controller = new AbortController()
   132→    abortRef.current = controller
   133→
   134→    setLoading(true)
   135→    setError(null)
   136→    setFailedMission(null)
   137→    setChatMessages([])
   138→
   139→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   140→    const steps = extractStepsFromMission(m)
   141→    setBuildSteps(steps)
   142→    setPlanSummary(null)
   143→    setLivePreviewHtml(null)
   144→
   145→    // Helper: set error state consistently (replaces 6 repeated blocks)
   146→    const fail = (msg: string) => {
   147→      setError(msg)
   148→      setFailedMission(m)
   149→      if (!resultRef.current) toast.error(msg)
   150→    }
   151→
   152→    try {
   153→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   154→      // This is a separate, fast call (~2-3s). The plan lets us show
   155→      // AUTHENTIC steps to the user before the coder even starts.
   156→      const archRes = await fetch('/api/build/architect', {
   157→        method: 'POST',
   158→        headers: { 'Content-Type': 'application/json' },
   159→        body: JSON.stringify({ mission: m }),
   160→        signal: controller.signal,
   161→      })
   162→
   163→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   164→      const archContentType = archRes.headers.get('content-type') ?? ''
   165→      if (archContentType.includes('application/json')) {
   166→        try { archData = await archRes.json() } catch {}
   167→      }
   168→
   169→      // If architect succeeded, update steps with the REAL plan
   170→      if (archData?.ok && archData.plan) {
   171→        const planSteps = extractStepsFromPlan(archData.plan, m)
   172→        const summary = getPlanSummary(archData.plan)
   173→        setBuildSteps(planSteps)
   174→        setPlanSummary(summary)
   175→        console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   176→      }
   177→      // If architect failed, continue with mission-based steps (already set)
   178→
   179→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   180→      // The route returns Server-Sent Events: progress events while LLM works,
   181→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   182→      const codeRes = await fetch('/api/build/code', {
   183→        method: 'POST',
   184→        headers: { 'Content-Type': 'application/json' },
   185→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null }),
   186→        signal: controller.signal,
   187→      })
   188→
   189→      if (!codeRes.ok) {
   190→        // Non-SSE error (400, 429, 413) — parse as JSON
   191→        let errorMsg = `Server error (${codeRes.status})`
   192→        try {
   193→          const errData = await codeRes.json()
   194→          if (errData?.error) errorMsg = errData.error
   195→        } catch {}
   196→        fail(errorMsg)
   197→        return
   198→      }
   199→
   200→      // Read SSE stream
   201→      const reader = codeRes.body?.getReader()
   202→      if (!reader) {
   203→        fail('No response stream')
   204→        return
   205→      }
   206→
   207→      const decoder = new TextDecoder()
   208→      let buffer = ''
   209→      let finalHtml = ''
   210→      let finalTokens = 0
   211→      let finalMs = 0
   212→      let finalQuality = 0
   213→      let finalMetrics = ''
   214→      let streamError: string | null = null
   215→
   216→      while (true) {
   217→        const { done, value } = await reader.read()
   218→        if (done) break
   219→
   220→        buffer += decoder.decode(value, { stream: true })
   221→
   222→        // Parse SSE events (separated by \n\n)
   223→        const events = buffer.split('\n\n')
   224→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   225→
   226→        for (const eventStr of events) {
   227→          const dataLine = eventStr.trim()
   228→          if (!dataLine.startsWith('data: ')) continue
   229→
   230→          try {
   231→            const evt = JSON.parse(dataLine.slice(6))
   232→
   233→            if (evt.type === 'progress') {
   234→              // Update thinking steps with real progress from server
   235→              setBuildSteps(prev => {
   236→                // Replace with server-provided step, keep last as fallback
   237→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   238→                return [prev[0], evt.step, ...prev.slice(2)]
   239→              })
   240→            } else if (evt.type === 'token') {
   241→              // REAL TOKEN STREAMING — accumulate partial HTML for live preview
   242→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   243→              setBuildSteps(prev => {
   244→                const last = prev[prev.length - 1]
   245→                const totalLen = (evt.length ?? 0)
   246→                const newLast = `Generating: ${totalLen} chars...`
   247→                if (last && last.startsWith('Generating:')) {
   248→                  return [...prev.slice(0, -1), newLast]
   249→                }
   250→                return [...prev, newLast]
   251→              })
   252→            } else if (evt.type === 'result') {
   253→              finalHtml = evt.html ?? ''
   254→              finalTokens = evt.tokens ?? 0
   255→              finalMs = evt.ms ?? 0
   256→              finalQuality = evt.quality ?? 0
   257→              finalMetrics = evt.metrics ?? ''
   258→            } else if (evt.type === 'error') {
   259→              streamError = evt.error ?? 'Unknown error'
   260→            }
   261→          } catch {
   262→            // Ignore malformed events
   263→          }
   264→        }
   265→      }
   266→
   267→      if (streamError) {
   268→        fail(streamError)
   269→        return
   270→      }
   271→
   272→      if (!finalHtml) {
   273→        fail('Server returned empty HTML')
   274→        return
   275→      }
   276→
   277→      const buildResult: BuildResult = {
   278→        id: newBuildId(),
   279→        html: finalHtml,
   280→        tokens: finalTokens,
   281→        ms: finalMs,
   282→        mission: m,
   283→      }
   284→
   285→      setResult(buildResult)
   286→
   287→      // Functional setState — avoids stale closure on rapid successive builds
   288→      setHistory(prev => {
   289→        // Dedupe by mission (keep only the latest build per mission)
   290→        const next = [buildResult, ...prev.filter(h => h.mission !== m)].slice(0, 10)
   291→        // Best-effort localStorage; shrink if quota exceeded
   292→        let savedCount = next.length
   293→        try {
   294→          localStorage.setItem('nova_history', JSON.stringify(next))
   295→        } catch (quotaErr) {
   296→          console.error('[NOVA] localStorage quota exceeded:', quotaErr)
   297→          savedCount = 0
   298→          for (let i = next.length - 1; i >= 0; i--) {
   299→            try {
   300→              localStorage.setItem('nova_history', JSON.stringify(next.slice(0, i + 1)))
   301→              savedCount = i + 1
   302→              break
   303→            } catch {
   304→              // keep trying smaller
   305→            }
   306→          }
   307→        }
   308→        // Warn if we couldn't save everything
   309→        if (savedCount < next.length) {
   310→          toast.error(`localStorage full — only ${savedCount} of ${next.length} builds saved to history`)
   311→        }
   312→        return next
   313→      })
   314→
   315→      toast.success(`Built in ${(finalMs / 1000).toFixed(1)}s · ${finalTokens} tokens · quality: ${finalQuality}`)
   316→      setQualityScore(finalQuality)
   317→      setQualityMetrics(finalMetrics)
   318→    } catch (err: unknown) {
   319→      // AbortError = user started a new build, loaded history, or navigated away; silently ignore
   320→      if (err instanceof DOMException && err.name === 'AbortError') return
   321→      const msg = err instanceof Error ? err.message : 'Network error'
   322→      fail(msg)
   323→    } finally {
   324→      // Only clear loading if this controller is still the active one
   325→      if (abortRef.current === controller) {
   326→        abortRef.current = null
   327→        setLoading(false)
   328→      }
   329→    }
   330→  }, [mission])
   331→
   332→  const loadFromHistory = useCallback((h: BuildResult) => {
   333→    // Abort any in-flight build or refine
   334→    abortRef.current?.abort()
   335→    abortRef.current = null
   336→    refineAbortRef.current?.abort()
   337→    refineAbortRef.current = null
   338→    setLoading(false)
   339→    setRefining(false)
   340→    setResult(h)
   341→    setMission(h.mission)
   342→    setError(null)
   343→    setFailedMission(null)
   344→    setChatMessages([])
   345→  }, [])
   346→
   347→  const cancelBuild = useCallback(() => {
   348→    // Cancel only aborts the in-flight build — does NOT clear mission or result.
   349→    // The user keeps their mission text and the old preview (if any).
   350→    abortRef.current?.abort()
   351→    abortRef.current = null
   352→    setLoading(false)
   353→    setError(null)
   354→    setFailedMission(null)
   355→  }, [])
   356→
   357→  const reset = useCallback(() => {
   358→    // Reset clears everything — used by the "New" button.
   359→    abortRef.current?.abort()
   360→    abortRef.current = null
   361→    setLoading(false)
   362→    setRefining(false)
   363→    setResult(null)
   364→    setError(null)
   365→    setFailedMission(null)
   366→    setMission('')
   367→    setChatMessages([])
   368→    setQualityScore(0)
   369→    setQualityMetrics('')
   370→    setLivePreviewHtml(null)
   371→  }, [])
   372→
   373→  const retryFailed = useCallback(() => {
   374→    // If the user edited the textarea after the failure, use the edited mission.
   375→    // Otherwise, use the original failed mission.
   376→    const currentMission = mission.trim()
   377→    if (currentMission && currentMission !== failedMission) {
   378→      build(currentMission)
   379→    } else if (failedMission) {
   380→      build(failedMission)
   381→    }
   382→  }, [failedMission, build, mission])
   383→
   384→  const download = useCallback(() => {
   385→    if (!result?.html) return
   386→    const blob = new Blob([result.html], { type: 'text/html' })
   387→    const url = URL.createObjectURL(blob)
   388→    const a = document.createElement('a')
   389→    a.href = url
   390→    const filename = sanitizeFilename(result.mission)
   391→    a.download = filename
   392→    document.body.appendChild(a)
   393→    a.click()
   394→    document.body.removeChild(a)
   395→    URL.revokeObjectURL(url)
   396→    toast.success(`Downloaded ${filename}`)
   397→  }, [result])
   398→
   399→  // Chat refine: SSE streaming — same pattern as build/code
   400→  const sendChat = useCallback(async () => {
   401→    const msg = chatInput.trim()
   402→    const currentResult = resultRef.current
   403→    if (!msg || refining || !currentResult) return
   404→
   405→    const userMsg: ChatMessage = { role: 'user', content: msg, ts: Date.now() }
   406→    setChatMessages(prev => [...prev, userMsg])
   407→    setChatInput('')
   408→    setRefining(true)
   409→
   410→    refineAbortRef.current?.abort()
   411→    const controller = new AbortController()
   412→    refineAbortRef.current = controller
   413→
   414→    try {
   415→      const res = await fetch('/api/refine', {
   416→        method: 'POST',
   417→        headers: { 'Content-Type': 'application/json' },
   418→        body: JSON.stringify({ mission: currentResult.mission, html: currentResult.html, message: msg }),
   419→        signal: controller.signal,
   420→      })
   421→
   422→      if (!res.ok) {
   423→        let errorMsg = `Server error (${res.status})`
   424→        try { const e = await res.json(); if (e?.error) errorMsg = e.error } catch {}
   425→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${errorMsg}`, ts: Date.now() }])
   426→        toast.error(errorMsg)
   427→        return
   428→      }
   429→
   430→      // Read SSE stream
   431→      const reader = res.body?.getReader()
   432→      if (!reader) {
   433→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: No stream', ts: Date.now() }])
   434→        return
   435→      }
   436→
   437→      const decoder = new TextDecoder()
   438→      let buffer = ''
   439→      let finalHtml = ''
   440→      let finalTokens = 0
   441→      let finalMs = 0
   442→      let finalQuality = 0
   443→      let finalMetrics = ''
   444→      let streamError: string | null = null
   445→
   446→      while (true) {
   447→        const { done, value } = await reader.read()
   448→        if (done) break
   449→
   450→        buffer += decoder.decode(value, { stream: true })
   451→        const events = buffer.split('\n\n')
   452→        buffer = events.pop() ?? ''
   453→
   454→        for (const eventStr of events) {
   455→          const dataLine = eventStr.trim()
   456→          if (!dataLine.startsWith('data: ')) continue
   457→          try {
   458→            const evt = JSON.parse(dataLine.slice(6))
   459→            if (evt.type === 'result') {
   460→              finalHtml = evt.html ?? ''
   461→              finalTokens = evt.tokens ?? 0
   462→              finalMs = evt.ms ?? 0
   463→              finalQuality = evt.quality ?? 0
   464→              finalMetrics = evt.metrics ?? ''
   465→            } else if (evt.type === 'error') {
   466→              streamError = evt.error ?? 'Unknown error'
   467→            }
   468→          } catch {}
   469→        }
   470→      }
   471→
   472→      if (streamError) {
   473→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${streamError}`, ts: Date.now() }])
   474→        toast.error(streamError)
   475→        return
   476→      }
   477→
   478→      if (!finalHtml) {
   479→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: Empty response', ts: Date.now() }])
   480→        return
   481→      }
   482→
   483→      const refinedResult: BuildResult = {
   484→        ...currentResult,
   485→        id: newBuildId(),
   486→        html: finalHtml,
   487→      }
   488→      setResult(refinedResult)
   489→
   490→      setHistory(prev => {
   491→        const next = [refinedResult, ...prev.filter(h => h.mission !== currentResult.mission)].slice(0, 10)
   492→        try { localStorage.setItem('nova_history', JSON.stringify(next)) } catch {}
   493→        return next
   494→      })
   495→
   496→      setChatMessages(prev => [...prev, {
   497→        role: 'assistant',
   498→        content: `Updated! ${finalMs ? `${(finalMs / 1000).toFixed(1)}s` : ''} · ${finalTokens} tokens`,
   499→        ts: Date.now(),
   500→      }])
   501→
   502→      toast.success(`Refined! ${(finalMs / 1000).toFixed(1)}s · quality: ${finalQuality}`)
   503→      setQualityScore(finalQuality)
   504→      setQualityMetrics(finalMetrics)
   505→    } catch (err: unknown) {
   506→      if (err instanceof DOMException && err.name === 'AbortError') return
   507→      const failMsg = err instanceof Error ? err.message : 'Network error'
   508→      setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${failMsg}`, ts: Date.now() }])
   509→      toast.error(failMsg)
   510→    } finally {
   511→      if (refineAbortRef.current === controller) {
   512→        refineAbortRef.current = null
   513→        setRefining(false)
   514→      }
   515→    }
   516→  }, [chatInput, refining])
   517→
   518→  // Auto-scroll chat to bottom on new messages
   519→  useEffect(() => {
   520→    if (chatScrollRef.current) {
   521→      chatScrollRef.current.scrollTop = chatScrollRef.current.scrollHeight
   522→    }
   523→  }, [chatMessages, refining])
   524→
   525→  // Keyboard shortcuts: Esc=cancel build/refine, ⌘S/Ctrl+S=download
   526→  useEffect(() => {
   527→    const onKey = (e: KeyboardEvent) => {
   528→      // Esc cancels a build or refine
   529→      if (e.key === 'Escape' && (loading || refining)) {
   530→        e.preventDefault()
   531→        if (loading) {
   532→          cancelBuild()
   533→          toast.info('Build cancelled')
   534→        } else if (refining) {
   535→          refineAbortRef.current?.abort()
   536→          refineAbortRef.current = null
   537→          setRefining(false)
   538→          toast.info('Refine cancelled')
   539→        }
   540→        return
   541→      }
   542→      // ⌘S / Ctrl+S downloads the current result (always preventDefault to stop browser save)
   543→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 's') {
   544→        e.preventDefault()
   545→        if (result) {
   546→          download()
   547→        }
   548→        return
   549→      }
   550→    }
   551→    window.addEventListener('keydown', onKey)
   552→    return () => window.removeEventListener('keydown', onKey)
   553→  }, [loading, refining, result, download, cancelBuild])
   554→
   555→  // Helper: get current thinking step text (DYNAMIC — from mission or plan)
   556→  const getThinkingText = useCallback(() => {
   557→    if (loading) {
   558→      return buildSteps[thinkingStep] ?? buildSteps[buildSteps.length - 1] ?? 'Building...'
   559→    }
   560→    if (refining) {
   561→      return REFINE_THINKING_STEPS[thinkingStep] ?? 'Refining...'
   562→    }
   563→    return ''
   564→  }, [loading, refining, thinkingStep, buildSteps])
   565→
   566→  // Whether to show examples (only when no result, no error, not loading)
   567→  const showExamples = !result && !loading && !error
   568→  // Whether to show first-build error panel (no result, has error, not loading)
   569→  const showFirstError = !result && !!error && !loading
   570→
