   121→   121→  // Centralized build function. Aborts any in-flight build first.

   123→   123→  // Uses resultRef instead of result state to avoid being re-created on every build.
   124→   124→  const build = useCallback(async (explicitMission?: string) => {
   125→   125→    const m = (explicitMission ?? mission).trim()
   126→   126→    if (!m) {
   127→   127→      toast.error('Describe what to build first')
   128→   128→      return
   129→   129→    }
   130→   130→
   131→   131→    // If an explicit mission was passed, sync the textarea state

   133→   133→      setMission(explicitMission)
   134→   134→    }
   135→   135→
   136→   136→    // Abort any in-flight build (covers: rebuild, history-click-during-build, reset-during-build)
   137→   137→    abortRef.current?.abort()
   138→   138→    const controller = new AbortController()
   139→   139→    abortRef.current = controller
   140→   140→
   141→   141→    setLoading(true)
   142→   142→    setError(null)
   143→   143→    setFailedMission(null)
   144→   144→    setChatMessages([])
   145→   145→
   146→   146→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   147→   147→    const steps = extractStepsFromMission(m)
   148→   148→    setBuildSteps(steps)
   149→   149→    setPlanSummary(null)
   150→   150→    setLivePreviewHtml(null)
   151→   151→
   152→   152→    // Helper: set error state consistently (replaces 6 repeated blocks)
   153→   153→    const fail = (msg: string) => {
   154→   154→      setError(msg)
   155→   155→      setFailedMission(m)
   156→   156→      if (!resultRef.current) toast.error(msg)
   157→   157→    }
   158→   158→
   159→   159→    try {
   160→   160→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   161→   161→      // This is a separate, fast call (~2-3s). The plan lets us show
   162→   162→      // AUTHENTIC steps to the user before the coder even starts.
   163→   163→      const archRes = await fetch('/api/build/architect', {
   164→   164→        method: 'POST',
   165→   165→        headers: { 'Content-Type': 'application/json' },
   166→   166→        body: JSON.stringify({ mission: m }),
   167→   167→        signal: controller.signal,
   168→   168→      })
   169→   169→
   170→   170→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   171→   171→      const archContentType = archRes.headers.get('content-type') ?? ''
   172→   172→      if (archContentType.includes('application/json')) {
   173→   173→        try { archData = await archRes.json() } catch {}
   174→   174→      }
   175→   175→
   176→   176→      // If architect succeeded, update steps with the REAL plan
   177→   177→      if (archData?.ok && archData.plan) {
   178→   178→        const planSteps = extractStepsFromPlan(archData.plan, m)
   179→   179→        const summary = getPlanSummary(archData.plan)
   180→   180→        setBuildSteps(planSteps)
   181→   181→        setPlanSummary(summary)
   182→   182→        console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   183→   183→      }
   184→   184→      // If architect failed, continue with mission-based steps (already set)
   185→   185→
   186→   186→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   187→   187→      // The route returns Server-Sent Events: progress events while LLM works,
   188→   188→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   189→   189→      const codeRes = await fetch('/api/build/code', {
   190→   190→        method: 'POST',
   191→   191→        headers: { 'Content-Type': 'application/json' },
   192→   192→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null }),
   193→   193→        signal: controller.signal,
   194→   194→      })
   195→   195→
   196→   196→      if (!codeRes.ok) {
   197→   197→        // Non-SSE error (400, 429, 413) — parse as JSON
   198→   198→        let errorMsg = `Server error (${codeRes.status})`
   199→   199→        try {
   200→   200→          const errData = await codeRes.json()
   201→   201→          if (errData?.error) errorMsg = errData.error
   202→   202→        } catch {}
   203→   203→        fail(errorMsg)
   204→   204→        return
   205→   205→      }
   206→   206→
   207→   207→      // Read SSE stream
   208→   208→      const reader = codeRes.body?.getReader()
   209→   209→      if (!reader) {
   210→   210→        fail('No response stream')
   211→   211→        return
   212→   212→      }
   213→   213→
   214→   214→      const decoder = new TextDecoder()
   215→   215→      let buffer = ''
   216→   216→      let finalHtml = ''
   217→   217→      let finalTokens = 0
   218→   218→      let finalMs = 0
   219→   219→      let finalQuality = 0
   220→   220→      let finalMetrics = ''
   221→   221→      let streamError: string | null = null
   222→   222→
   223→   223→      while (true) {
   224→   224→        const { done, value } = await reader.read()
   225→   225→        if (done) break
   226→   226→
   227→   227→        buffer += decoder.decode(value, { stream: true })
   228→   228→
   229→   229→        // Parse SSE events (separated by \n\n)
   230→   230→        const events = buffer.split('\n\n')
   231→   231→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   232→   232→
   233→   233→        for (const eventStr of events) {
   234→   234→          const dataLine = eventStr.trim()
   235→   235→          if (!dataLine.startsWith('data: ')) continue
   236→   236→
   237→   237→          try {
   238→   238→            const evt = JSON.parse(dataLine.slice(6))
   239→   239→
   240→   240→            if (evt.type === 'progress') {
   241→   241→              // Update thinking steps with real progress from server
   242→   242→              setBuildSteps(prev => {
   243→   243→                // Replace with server-provided step, keep last as fallback
   244→   244→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   245→   245→                return [prev[0], evt.step, ...prev.slice(2)]
   246→   246→              })
   247→   247→            } else if (evt.type === 'token') {
   248→   248→              // REAL TOKEN STREAMING — accumulate partial HTML for live preview
   249→   249→              setLivePreviewHtml(prev => (prev ?? '') + (evt.text ?? ''))
   250→   250→              setBuildSteps(prev => {
   251→   251→                const last = prev[prev.length - 1]
   252→   252→                const totalLen = (evt.length ?? 0)
   253→   253→                const newLast = `Generating: ${totalLen} chars...`
   254→   254→                if (last && last.startsWith('Generating:')) {
   255→   255→                  return [...prev.slice(0, -1), newLast]
   256→   256→                }
   257→   257→                return [...prev, newLast]
   258→   258→              })
   259→   259→            } else if (evt.type === 'result') {
   260→   260→              finalHtml = evt.html ?? ''
   261→   261→              finalTokens = evt.tokens ?? 0
   262→   262→              finalMs = evt.ms ?? 0
   263→   263→              finalQuality = evt.quality ?? 0
   264→   264→              finalMetrics = evt.metrics ?? ''
   265→   265→            } else if (evt.type === 'error') {
   266→   266→              streamError = evt.error ?? 'Unknown error'
   267→   267→            }
   268→   268→          } catch {
   269→   269→            // Ignore malformed events
   270→   270→          }
   271→   271→        }
   272→   272→      }
   273→   273→
   274→   274→      if (streamError) {
   275→   275→        fail(streamError)
   276→   276→        return
   277→   277→      }
   278→   278→
   279→   279→      if (!finalHtml) {
   280→   280→        fail('Server returned empty HTML')
   281→   281→        return
   282→   282→      }
   283→   283→
   284→   284→      const buildResult: BuildResult = {
   285→   285→        id: newBuildId(),
   286→   286→        html: finalHtml,
   287→   287→        tokens: finalTokens,
   288→   288→        ms: finalMs,
   289→   289→        mission: m,
   290→   290→      }
   291→   291→
   292→   292→      setResult(buildResult)
   293→   293→
   294→   294→      // Functional setState — avoids stale closure on rapid successive builds
   295→   295→      setHistory(prev => {
   296→   296→        // Dedupe by mission (keep only the latest build per mission)
   297→   297→        const next = [buildResult, ...prev.filter(h => h.mission !== m)].slice(0, 10)
   298→   298→        // Best-effort localStorage; shrink if quota exceeded
   299→   299→        let savedCount = next.length
   300→   300→        try {
   301→   301→          localStorage.setItem('nova_history', JSON.stringify(next))
   302→   302→        } catch (quotaErr) {
   303→   303→          console.error('[NOVA] localStorage quota exceeded:', quotaErr)
   304→   304→          savedCount = 0
   305→   305→          for (let i = next.length - 1; i >= 0; i--) {
   306→   306→            try {
   307→   307→              localStorage.setItem('nova_history', JSON.stringify(next.slice(0, i + 1)))
   308→   308→              savedCount = i + 1
   309→   309→              break
   310→   310→            } catch {
   311→   311→              // keep trying smaller
   312→   312→            }
   313→   313→          }
   314→   314→        }
   315→   315→        // Warn if we couldn't save everything
   316→   316→        if (savedCount < next.length) {
   317→   317→          toast.error(`localStorage full — only ${savedCount} of ${next.length} builds saved to history`)
   318→   318→        }
   319→   319→        return next
   320→   320→      })
   321→   321→
   322→   322→      toast.success(`Built in ${(finalMs / 1000).toFixed(1)}s · ${finalTokens} tokens · quality: ${finalQuality}`)
   323→   323→      setQualityScore(finalQuality)
   324→   324→      setQualityMetrics(finalMetrics)
   325→   325→    } catch (err: unknown) {
   326→   326→      // AbortError = user started a new build, loaded history, or navigated away; silently ignore
   327→   327→      if (err instanceof DOMException && err.name === 'AbortError') return
   328→   328→      const msg = err instanceof Error ? err.message : 'Network error'
   329→   329→      fail(msg)
   330→   330→    } finally {
   331→   331→      // Only clear loading if this controller is still the active one
   332→   332→      if (abortRef.current === controller) {
   333→   333→        abortRef.current = null
   334→   334→        setLoading(false)
   335→   335→      }
   336→   336→    }
   337→   337→  }, [mission])
   338→   338→
   339→   339→  const loadFromHistory = useCallback((h: BuildResult) => {
   340→   340→    // Abort any in-flight build or refine
   341→   341→    abortRef.current?.abort()
   342→   342→    abortRef.current = null
   343→   343→    refineAbortRef.current?.abort()
   344→   344→    refineAbortRef.current = null
   345→   345→    setLoading(false)
   346→   346→    setRefining(false)
   347→   347→    setResult(h)
   348→   348→    setMission(h.mission)
   349→   349→    setError(null)
   350→   350→    setFailedMission(null)

   380→   380→  const retryFailed = useCallback(() => {
   381→   381→    // If the user edited the textarea after the failure, use the edited mission.

   383→   383→    const currentMission = mission.trim()
   384→   384→    if (currentMission && currentMission !== failedMission) {

   390→   390→
   391→   391→  const download = useCallback(() => {
   392→   392→    if (!result?.html) return
   393→   393→    const blob = new Blob([result.html], { type: 'text/html' })
   394→   394→    const url = URL.createObjectURL(blob)

   398→   398→    a.download = filename
   399→   399→    document.body.appendChild(a)
   400→   400→    a.click()
   401→   401→    document.body.removeChild(a)
   402→   402→    URL.revokeObjectURL(url)
   403→   403→    toast.success(`Downloaded ${filename}`)
   404→   404→  }, [result])
   405→   405→
   406→   406→  // Copy HTML to clipboard
   407→   407→  const copyHtml = useCallback(async () => {
   408→   408→    if (!result?.html) return
   409→   409→    try {
   410→   410→      await navigator.clipboard.writeText(result.html)
   411→   411→      toast.success('HTML copied to clipboard')
   412→   412→    } catch {
   413→   413→      toast.error('Failed to copy — try downloading instead')
   414→   414→    }
   415→   415→  }, [result])
   416→   416→
   417→   417→  // Open the HTML in a new browser tab (full-screen preview)
   418→   418→  const openInNewTab = useCallback(() => {
   419→   419→    if (!result?.html) return
   420→   420→    const blob = new Blob([result.html], { type: 'text/html' })
   421→   421→    const url = URL.createObjectURL(blob)
   422→   422→    window.open(url, '_blank', 'noopener,noreferrer')
   423→   423→    // Revoke after 30s — enough time for the tab to load
   424→   424→    setTimeout(() => URL.revokeObjectURL(url), 30_000)
   425→   425→    toast.info('Opened in new tab')
   426→   426→  }, [result])
   427→   427→
   428→   428→  // Chat refine: SSE streaming — same pattern as build/code
   429→   429→  const sendChat = useCallback(async () => {
   430→   430→    const msg = chatInput.trim()
   431→   431→    const currentResult = resultRef.current
   432→   432→    if (!msg || refining || !currentResult) return
   433→   433→
   434→   434→    const userMsg: ChatMessage = { role: 'user', content: msg, ts: Date.now() }
   435→   435→    setChatMessages(prev => [...prev, userMsg])
   436→   436→    setChatInput('')
   437→   437→    setRefining(true)
   438→   438→
   439→   439→    refineAbortRef.current?.abort()
   440→   440→    const controller = new AbortController()
   441→   441→    refineAbortRef.current = controller
   442→   442→
   443→   443→    // Clear live preview so refine starts fresh
   444→   444→    setLivePreviewHtml(null)
   445→   445→
   446→   446→    try {
   447→   447→      const res = await fetch('/api/refine', {
   448→   448→        method: 'POST',
   449→   449→        headers: { 'Content-Type': 'application/json' },
   450→   450→        body: JSON.stringify({ mission: currentResult.mission, html: currentResult.html, message: msg }),
   451→   451→        signal: controller.signal,
   452→   452→      })
   453→   453→
   454→   454→      if (!res.ok) {
   455→   455→        let errorMsg = `Server error (${res.status})`
   456→   456→        try { const e = await res.json(); if (e?.error) errorMsg = e.error } catch {}
   457→   457→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${errorMsg}`, ts: Date.now() }])
   458→   458→        toast.error(errorMsg)
   459→   459→        return
   460→   460→      }
   461→   461→
   462→   462→      // Read SSE stream
   463→   463→      const reader = res.body?.getReader()
   464→   464→      if (!reader) {
   465→   465→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: No stream', ts: Date.now() }])
   466→   466→        return
   467→   467→      }
   468→   468→
   469→   469→      const decoder = new TextDecoder()
   470→   470→      let buffer = ''
