    23→const mockFetch = mock((_url: string, _init?: unknown): Promise<Response> => {

    27→// Save the original fetch so we can restore it after the test file finishes.
    28→// This prevents the mock from leaking into other test files when running in
    29→// sequential (non --parallel) mode.
    30→const ORIGINAL_FETCH = globalThis.fetch
    31→
    32→// Install the mock on globalThis.fetch BEFORE importing the module under test.
    33→// Bun reads globalThis.fetch at call time, so installing before each test is
    34→// sufficient. But to ensure the mock is in place even if the module captures
    35→// fetch at load time, we install it once here.
    36→;(globalThis as unknown as { fetch: typeof mockFetch }).fetch = mockFetch
    37→
    38→// Restore the original fetch after all tests in this file complete. This is

    40→// default `bun test` does not).
    41→import { afterAll } from 'bun:test'
    42→afterAll(() => {

    57→function makeResponse(opts: MockResponseInit): Response {
    58→  const status = opts.status ?? 200
    59→  // Explicit null body — Response(null) creates a Response with body=null.
    60→  if (opts.body === null) {
    61→    return new Response(null, { status })
    62→  }
    63→  if (opts.body) {
    64→    return new Response(opts.body, { status })
    65→  }
    66→  if (opts.json) {
    67→    const body = JSON.stringify(opts.json())
    68→    return new Response(body, { status, headers: { 'Content-Type': 'application/json' } })
    69→  }
    70→  if (opts.text !== undefined) {
    71→    return new Response(opts.text, { status })
    72→  }
    73→  return new Response('', { status })
    74→}

    76→function makeSseBody(events: Array<string>): ReadableStream<Uint8Array> {
    77→  const encoder = new TextEncoder()
    78→  return new ReadableStream<Uint8Array>({
    79→    start(controller) {
    80→      for (const e of events) controller.enqueue(encoder.encode(e))
    81→      controller.close()
    82→    },
    83→  })
    84→}

    86→/** Build an OpenAI-compatible SSE event string. */
    87→function sse(obj: unknown): string {
    88→  return `data: ${JSON.stringify(obj)}\n\n`
    89→}

   583→describe('tokenRouterChat — abort signal', () => {
   584→  it('returns "cancelled" when external signal is already aborted', async () => {
   585→    const controller = new AbortController()
   586→    controller.abort()
   587→
   588→    // fetch will reject because the signal is aborted.
   589→    mockFetch.mockImplementation((_url, init) => {
   590→      const i = init as { signal?: AbortSignal }
