    71→  // Ref mirror of `result` so build() doesn't need it in useCallback deps.
    72→  // This prevents build from being re-created on every result change (every build).
    73→  // Updated in a useEffect (not during render — that's a side effect).

    79→  // Throttled live-preview accumulator.
    80→  // CRITICAL: Without throttling, setLivePreviewHtml fires on every token (~2000+ times per build),
    81→  // and each fires a full iframe srcDoc reload (re-parse + re-render + re-run partial scripts).
    82→  // We accumulate tokens in a ref and flush to state at most every 200ms via an interval.
    83→  const livePreviewAccumulatorRef = useRef<string>('')
    84→  const livePreviewFlushTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    85→
    86→  // Start the throttled flush timer when loading or refining begins; clear when it ends.
    87→  useEffect(() => {
    88→    if (!loading && !refining) {
    89→      // Flush any remaining accumulated text, then stop the timer.
    90→      if (livePreviewAccumulatorRef.current) {
    91→        setLivePreviewHtml(livePreviewAccumulatorRef.current)
    92→        livePreviewAccumulatorRef.current = ''
    93→      }
    94→      if (livePreviewFlushTimerRef.current) {
    95→        clearInterval(livePreviewFlushTimerRef.current)
    96→        livePreviewFlushTimerRef.current = null
    97→      }
    98→      return
    99→    }
   100→    // Start a 200ms flush interval — updates the iframe at most 5 times/second instead of ~2000.
   101→    if (!livePreviewFlushTimerRef.current) {
   102→      livePreviewFlushTimerRef.current = setInterval(() => {
   103→        if (livePreviewAccumulatorRef.current) {
   104→          setLivePreviewHtml(livePreviewAccumulatorRef.current)
   105→          // Don't clear the accumulator — new tokens may have arrived since the flush.
   106→          // It's overwritten on the next token event.
   107→        }
   108→      }, 200)
   109→    }
   110→    return () => {
   111→      // Cleanup on unmount or dep change — don't leave the interval running.
   112→      if (livePreviewFlushTimerRef.current) {
   113→        clearInterval(livePreviewFlushTimerRef.current)
   114→        livePreviewFlushTimerRef.current = null
   115→      }
   116→    }
   117→  }, [loading, refining])
   118→
   119→  // Load history from localStorage on mount
   120→  useEffect(() => {
   121→    try {
   122→      const stored = JSON.parse(localStorage.getItem('nova_history') ?? '[]')
   123→      setHistory(validateHistory(stored))
   124→    } catch (err) {
   125→      console.error('[NOVA] Failed to load history:', err)
   126→    }
   127→  }, [])
   128→
   129→  // Focus the mission textarea on mount — desktop only.
   130→  // On mobile, auto-focusing pops the on-screen keyboard which is annoying.
   131→  // Runs after mount to avoid SSR hydration mismatch.
   132→  useEffect(() => {
   133→    if (typeof window !== 'undefined' && window.matchMedia('(min-width: 768px)').matches) {
   134→      document.getElementById('mission-input')?.focus()
   135→    }
   136→  }, [])
   137→
   138→  // Save history to localStorage — pure side effect, called OUTSIDE of setState updaters
   139→  // to avoid double-firing in React StrictMode.
   140→  const saveHistoryToStorage = useCallback((items: BuildResult[]) => {
   141→    let savedCount = items.length
   142→    try {
   143→      localStorage.setItem('nova_history', JSON.stringify(items))
   144→    } catch (quotaErr) {
   145→      console.error('[NOVA] localStorage quota exceeded:', quotaErr)
   146→      savedCount = 0
   147→      for (let i = items.length - 1; i >= 0; i--) {
   148→        try {
   149→          localStorage.setItem('nova_history', JSON.stringify(items.slice(0, i + 1)))
   150→          savedCount = i + 1
   151→          break
   152→        } catch {
   153→          // keep trying smaller
   154→        }
   155→      }
   156→    }
   157→    if (savedCount < items.length) {
   158→      toast.error(`localStorage full — only ${savedCount} of ${items.length} builds saved to history`)
   159→    }
   160→  }, [])
   161→
   162→  // Add a build result to history (dedupe by mission) and persist to localStorage.
   163→  // Side effects (localStorage, toast) happen here — NOT inside the setState updater.
   164→  const addBuildToHistory = useCallback((buildResult: BuildResult) => {
   165→    let newHistory: BuildResult[] = []
   166→    setHistory(prev => {
   167→      newHistory = [buildResult, ...prev.filter(h => h.mission !== buildResult.mission)].slice(0, 10)
   168→      return newHistory
   169→    })
   170→    // Persist after state is computed (not inside updater — avoids StrictMode double-fire)
   171→    saveHistoryToStorage(newHistory)
   172→  }, [saveHistoryToStorage])
   173→
   174→  // Abort any in-flight build AND refine on unmount
   175→  useEffect(() => {
   176→    return () => {
   177→      abortRef.current?.abort()
   178→      refineAbortRef.current?.abort()
   179→    }
   180→  }, [])
   181→
   182→  // Ref mirror of buildSteps so the elapsed-time effect doesn't need buildSteps in its deps.
   183→  // If buildSteps is in the deps, the effect tears down and re-runs on every token event
   184→  // (because setBuildSteps is called on every token), resetting startTime and thinkingStep.
   185→  const buildStepsRef = useRef<string[]>(['Building...'])
   186→  useEffect(() => {
   187→    buildStepsRef.current = buildSteps
   188→  }, [buildSteps])
   189→
   190→  // Elapsed time counter + dynamic thinking step rotation.
   191→  // CRITICAL: deps are [loading, refining] ONLY — not buildSteps.
   192→  // If buildSteps were in deps, every token event would reset startTime to Date.now(),
   193→  // the interval would be cleared before it ever fires, and elapsed would stay at 0.
   194→  useEffect(() => {
   195→    if (!loading && !refining) {
   196→      setElapsed(0)
   197→      setThinkingStep(0)
   198→      return
   199→    }
   200→    const startTime = Date.now()
   201→    let step = 0
   202→    setThinkingStep(0)
   203→
   204→    const timer = setInterval(() => {
   205→      const sec = Math.floor((Date.now() - startTime) / 1000)
   206→      setElapsed(sec)
   207→
   208→      // Read steps from ref (not from closure — closure would be stale)
   209→      const steps = loading ? buildStepsRef.current : REFINE_THINKING_STEPS
   210→      if (steps.length === 0) return
   211→
   212→      // Rotate through steps — ~4s per step for build, 5s for refine
   213→      const stepDuration = loading ? 4 : 5
   214→      const nextStep = Math.min(steps.length - 1, Math.floor(sec / stepDuration))
   215→      if (nextStep !== step) {
   216→        step = nextStep
   217→        setThinkingStep(step)
   218→      }
   219→    }, 1000)
   220→    return () => clearInterval(timer)
   221→  }, [loading, refining])
   222→
   223→  // Centralized build function. Aborts any in-flight build first.
   224→  // Accepts an optional explicit mission to avoid stale-closure bugs (e.g., retry).
   225→  // Uses resultRef instead of result state to avoid being re-created on every build.
   226→  const build = useCallback(async (explicitMission?: string) => {
   227→    const m = (explicitMission ?? mission).trim()
   228→    if (!m) {
   229→      toast.error('Describe what to build first')
   230→      return
   231→    }
   232→
   233→    // If an explicit mission was passed, sync the textarea state
   234→    if (explicitMission && explicitMission !== mission) {
   235→      setMission(explicitMission)
   236→    }
   237→
   238→    // Abort any in-flight build AND refine (covers: rebuild during refine, history-click-during-build, reset-during-build)
   239→    // This prevents the race condition where build and refine run simultaneously and
   240→    // both call setResult/setHistory, corrupting state.
   241→    abortRef.current?.abort()
   242→    refineAbortRef.current?.abort()
   243→    refineAbortRef.current = null
   244→    setRefining(false)
   245→    const controller = new AbortController()
   246→    abortRef.current = controller
   247→
   248→    setLoading(true)
   249→    setError(null)
   250→    setFailedMission(null)
   251→    setChatMessages([])
   252→    setConfirmClear(false) // Hide confirm-clear UI if it was visible
   253→
   254→    // Extract dynamic steps from the mission IMMEDIATELY — not pre-canned
   255→    const steps = extractStepsFromMission(m)
   256→    setBuildSteps(steps)
   257→    setPlanSummary(null)
   258→    setLivePreviewHtml(null)
   259→    livePreviewAccumulatorRef.current = '' // Clear accumulator for fresh build
   260→
   261→    // Helper: set error state consistently (replaces 6 repeated blocks)
   262→    const fail = (msg: string) => {
   263→      setError(msg)
   264→      setFailedMission(m)
   265→      if (!resultRef.current) toast.error(msg)
   266→    }
   267→
   268→    try {
   269→      // ═══ STAGE 1: ARCHITECT — get the plan first ═══
   270→      // This is a separate, fast call (~2-3s). The plan lets us show
   271→      // AUTHENTIC steps to the user before the coder even starts.
   272→      const archRes = await fetch('/api/build/architect', {
   273→        method: 'POST',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({ mission: m }),
   276→        signal: controller.signal,
   277→      })
   278→
   279→      let archData: { ok: boolean; plan?: unknown; error?: string } | null = null
   280→      const archContentType = archRes.headers.get('content-type') ?? ''
   281→      if (archContentType.includes('application/json')) {
   282→        try { archData = await archRes.json() } catch {}
   283→      }
   284→
   285→      // If architect succeeded, update steps with the REAL plan
   286→      if (archData?.ok && archData.plan) {
   287→        const planSteps = extractStepsFromPlan(archData.plan, m)
   288→        const summary = getPlanSummary(archData.plan)
   289→        setBuildSteps(planSteps)
   290→        setPlanSummary(summary)
   291→        // Debug log gated to non-production — don't spam user's console
   292→        if (process.env.NODE_ENV !== 'production') {
   293→          console.log('[NOVA] Architect plan:', summary, planSteps.length, 'steps')
   294→        }
   295→      }
   296→      // If architect failed, continue with mission-based steps (already set)
   297→
   298→      // ═══ STAGE 2: CODER — SSE streaming with keepalive ═══
   299→      // The route returns Server-Sent Events: progress events while LLM works,
   300→      // then a result event with the final HTML. No timeout — keepalive prevents it.
   301→      const codeRes = await fetch('/api/build/code', {
   302→        method: 'POST',
   303→        headers: { 'Content-Type': 'application/json' },
   304→        body: JSON.stringify({ mission: m, plan: archData?.plan ?? null }),
   305→        signal: controller.signal,
   306→      })
   307→
   308→      if (!codeRes.ok) {
   309→        // Non-SSE error (400, 429, 413) — parse as JSON
   310→        let errorMsg = `Server error (${codeRes.status})`
   311→        try {
   312→          const errData = await codeRes.json()
   313→          if (errData?.error) errorMsg = errData.error
   314→        } catch {}
   315→        fail(errorMsg)
   316→        return
   317→      }
   318→
   319→      // Read SSE stream
   320→      const reader = codeRes.body?.getReader()
   321→      if (!reader) {
   322→        fail('No response stream')
   323→        return
   324→      }
   325→
   326→      const decoder = new TextDecoder()
   327→      let buffer = ''
   328→      let finalHtml = ''
   329→      let finalTokens = 0
   330→      let finalMs = 0
   331→      let finalQuality = 0
   332→      let finalMetrics = ''
   333→      let streamError: string | null = null
   334→
   335→      while (true) {
   336→        const { done, value } = await reader.read()
   337→        if (done) break
   338→
   339→        buffer += decoder.decode(value, { stream: true })
   340→
   341→        // Parse SSE events (separated by \n\n or \r\n\r\n — some proxies normalize to \r\n)
   342→        // Normalize \r\n to \n first so the split works regardless of proxy behavior.
   343→        const normalized = buffer.replace(/\r\n/g, '\n')
   344→        const events = normalized.split('\n\n')
   345→        buffer = events.pop() ?? '' // keep incomplete event in buffer
   346→
   347→        for (const eventStr of events) {
   348→          const dataLine = eventStr.trim()
   349→          if (!dataLine.startsWith('data: ')) continue
   350→
   351→          try {
   352→            const evt = JSON.parse(dataLine.slice(6))
   353→
   354→            if (evt.type === 'progress') {
   355→              // Update thinking steps with real progress from server
   356→              setBuildSteps(prev => {
   357→                // Replace with server-provided step, keep last as fallback
   358→                if (prev.length <= 2) return [evt.step, 'Generating code...', 'Finalizing...']
   359→                return [prev[0], evt.step, ...prev.slice(2)]
   360→              })
   361→            } else if (evt.type === 'token') {
   362→              // REAL TOKEN STREAMING — accumulate in ref for throttled flush.
   363→              // We DON'T call setLivePreviewHtml here (would reload iframe on every token).
   364→              // The 200ms flush interval picks up the accumulated text.
   365→              livePreviewAccumulatorRef.current = (livePreviewAccumulatorRef.current || '') + (evt.text ?? '')
   366→              setBuildSteps(prev => {
   367→                const last = prev[prev.length - 1]
   368→                const totalLen = (evt.length ?? 0)
   369→                const newLast = `Generating: ${totalLen} chars...`
   370→                if (last && last.startsWith('Generating:')) {
   371→                  return [...prev.slice(0, -1), newLast]
   372→                }
   373→                return [...prev, newLast]
   374→              })
   375→            } else if (evt.type === 'result') {
   376→              finalHtml = evt.html ?? ''
   377→              finalTokens = evt.tokens ?? 0
   378→              finalMs = evt.ms ?? 0
   379→              finalQuality = evt.quality ?? 0
   380→              finalMetrics = evt.metrics ?? ''
   381→            } else if (evt.type === 'error') {
   382→              streamError = evt.error ?? 'Unknown error'
   383→            }
   384→          } catch {
   385→            // Ignore malformed events
   386→          }
   387→        }
   388→      }
   389→
   390→      if (streamError) {
   391→        fail(streamError)
   392→        return
   393→      }
   394→
   395→      if (!finalHtml) {
   396→        fail('Server returned empty HTML')
   397→        return
   398→      }
   399→
   400→      const buildResult: BuildResult = {
   401→        id: newBuildId(),
   402→        html: finalHtml,
   403→        tokens: finalTokens,
   404→        ms: finalMs,
   405→        mission: m,
   406→      }
   407→
   408→      setResult(buildResult)
   409→      resultRef.current = buildResult // Update ref synchronously
   410→      addBuildToHistory(buildResult)
   411→
   412→      toast.success(`Built in ${(finalMs / 1000).toFixed(1)}s · ${finalTokens} tokens · quality: ${finalQuality}`)
   413→      setQualityScore(finalQuality)
   414→      setQualityMetrics(finalMetrics)
   415→    } catch (err: unknown) {
   416→      // AbortError = user started a new build, loaded history, or navigated away; silently ignore
   417→      if (err instanceof DOMException && err.name === 'AbortError') return
   418→      const msg = err instanceof Error ? err.message : 'Network error'
   419→      fail(msg)
   420→    } finally {
   421→      // Only clear loading if this controller is still the active one
   422→      if (abortRef.current === controller) {
   423→        abortRef.current = null
   424→        setLoading(false)
   425→      }
   426→    }
   427→  }, [mission])
   428→
   429→  const loadFromHistory = useCallback((h: BuildResult) => {
   430→    // Abort any in-flight build or refine
   431→    abortRef.current?.abort()
   432→    abortRef.current = null
   433→    refineAbortRef.current?.abort()
   434→    refineAbortRef.current = null
   435→    setLoading(false)
   436→    setRefining(false)
   437→    setResult(h)
   438→    resultRef.current = h // Update ref synchronously so sendChat sees the correct result immediately
   439→    setMission(h.mission)
   440→    setError(null)
   441→    setFailedMission(null)
   442→    setChatMessages([])
   443→    // Reset all derived state so we don't show the previous build's badges/plan
   444→    setQualityScore(0)
   445→    setQualityMetrics('')
   446→    setPlanSummary(null)
   447→    setLivePreviewHtml(null)
   448→    setConfirmClear(false)
   449→  }, [])
   450→
   451→  const cancelBuild = useCallback(() => {
   452→    // Cancel only aborts the in-flight build — does NOT clear mission or result.
   453→    // The user keeps their mission text and the old preview (if any).
   454→    abortRef.current?.abort()
   455→    abortRef.current = null
   456→    setLoading(false)
   457→    setError(null)
   458→    setFailedMission(null)
   459→    setLivePreviewHtml(null) // Clear stale partial HTML so it doesn't flash back later
   460→  }, [])
   461→
   462→  // Cancel an in-flight refine — accessible via the toolbar Cancel button during refining.
   463→  // Previously, the Cancel button only appeared during loading, leaving no mouse-accessible
   464→  // way to cancel a refine (only Esc worked, which is not discoverable).
   465→  const cancelRefine = useCallback(() => {
   466→    refineAbortRef.current?.abort()
   467→    refineAbortRef.current = null
   468→    setRefining(false)
   469→    setLivePreviewHtml(null)
   470→  }, [])
   471→
   472→  const reset = useCallback(() => {
   473→    // Reset clears everything — used by the "New" button.
   474→    // Abort BOTH in-flight build and refine to prevent phantom state mutations.
   475→    abortRef.current?.abort()
   476→    abortRef.current = null
   477→    refineAbortRef.current?.abort()
   478→    refineAbortRef.current = null
   479→    setLoading(false)
   480→    setRefining(false)
   481→    setResult(null)
   482→    setError(null)
   483→    setFailedMission(null)
   484→    setMission('')
   485→    setChatMessages([])
   486→    setQualityScore(0)
   487→    setQualityMetrics('')
   488→    setLivePreviewHtml(null)
   489→    setPlanSummary(null)
   490→    setConfirmClear(false)
   491→  }, [])
   492→
   493→  const retryFailed = useCallback(() => {
   494→    // If the user edited the textarea after the failure, use the edited mission.
   495→    // Otherwise, use the original failed mission.
   496→    const currentMission = mission.trim()
   497→    if (currentMission && currentMission !== failedMission) {
   498→      build(currentMission)
   499→    } else if (failedMission) {
   500→      build(failedMission)
   501→    }
   502→  }, [failedMission, build, mission])
   503→
   504→  const download = useCallback(() => {
   505→    if (!result?.html) return
   506→    const blob = new Blob([result.html], { type: 'text/html' })
   507→    const url = URL.createObjectURL(blob)
   508→    const a = document.createElement('a')
   509→    a.href = url
   510→    const filename = sanitizeFilename(result.mission)
   511→    a.download = filename
   512→    document.body.appendChild(a)
   513→    a.click()
   514→    document.body.removeChild(a)
   515→    // Delay revocation — Safari <16 and Firefox with large blobs read asynchronously.
   516→    // Revoking immediately can produce 0-byte files.
   517→    setTimeout(() => URL.revokeObjectURL(url), 1000)
   518→    toast.success(`Downloaded ${filename}`)
   519→  }, [result])
   520→
   521→  // Copy HTML to clipboard
   522→  const copyHtml = useCallback(async () => {
   523→    if (!result?.html) return
   524→    try {
   525→      await navigator.clipboard.writeText(result.html)
   526→      toast.success('HTML copied to clipboard')
   527→    } catch {
   528→      toast.error('Failed to copy — try downloading instead')
   529→    }
   530→  }, [result])
   531→
   532→  // Open the HTML in a new browser tab (full-screen preview)
   533→  const openInNewTab = useCallback(() => {
   534→    if (!result?.html) return
   535→    const blob = new Blob([result.html], { type: 'text/html' })
   536→    const url = URL.createObjectURL(blob)
   537→    window.open(url, '_blank', 'noopener,noreferrer')
   538→    // Revoke after 30s — enough time for the tab to load
   539→    setTimeout(() => URL.revokeObjectURL(url), 30_000)
   540→    toast.info('Opened in new tab')
   541→  }, [result])
   542→
   543→  // Chat refine: SSE streaming — same pattern as build/code
   544→  const sendChat = useCallback(async () => {

   602→
   603→        buffer += decoder.decode(value, { stream: true })

   605→        const normalized = buffer.replace(/\r\n/g, '\n')
   606→        const events = normalized.split('\n\n')

   610→          const dataLine = eventStr.trim()

   613→            const evt = JSON.parse(dataLine.slice(6))

   616→              livePreviewAccumulatorRef.current = (livePreviewAccumulatorRef.current || '') + (evt.text ?? '')

   620→              finalMs = evt.ms ?? 0
   621→              finalQuality = evt.quality ?? 0
   622→              finalMetrics = evt.metrics ?? ''
   623→            } else if (evt.type === 'error') {
   624→              streamError = evt.error ?? 'Unknown error'
   625→            }
   626→          } catch {}
   627→        }
   628→      }

   630→      if (streamError) {
   631→        setChatMessages(prev => [...prev, { role: 'assistant', content: `Error: ${streamError}`, ts: Date.now() }])
   632→        toast.error(streamError)
   633→        return
   634→      }
   635→
   636→      if (!finalHtml) {
   637→        setChatMessages(prev => [...prev, { role: 'assistant', content: 'Error: Empty response', ts: Date.now() }])
   638→        return
   639→      }
   640→
   641→      const refinedResult: BuildResult = {
   642→        ...currentResult,
   643→        id: newBuildId(),
   644→        html: finalHtml,
   645→        tokens: finalTokens, // Update tokens so the header shows the refine's token count
   646→        ms: finalMs,         // Update ms so the header shows the refine's time

   650→      addBuildToHistory(refinedResult)
   651→

   653→        role: 'assistant',
   654→        content: `Updated! ${finalMs ? `${(finalMs / 1000).toFixed(1)}s` : ''} · ${finalTokens} tokens`,
   655→        ts: Date.now(),
   656→      }])
